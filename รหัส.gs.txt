function initSheet() {
  throw new Error('Safety stop: initSheet() rebuilds the spreadsheet and can delete existing stock data. Use initSheet_CONFIRMED_RESET() only for a brand-new empty Sheet.');
}

function initSheet_CONFIRMED_RESET() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheets = ss.getSheets();
  const keeper = sheets[0];

  for (let i = 1; i < sheets.length; i++) {
    ss.deleteSheet(sheets[i]);
  }

  const wsDash = buildDashboard(ss, keeper);
  const wsDev  = buildDeviceSheet(ss);
  const wsMask = buildMaskSheet(ss);
  const wsAcc  = buildAccessorySheet(ss);
  const wsHist = buildHistorySheet(ss);
  const wsProd = buildProductDBSheet(ss);

  wsDash.setTabColor('#1a1a2e');
  wsDev.setTabColor('#1a4fa0');
  wsMask.setTabColor('#1a7a4a');
  wsAcc.setTabColor('#a06a00');
  wsHist.setTabColor('#4a3580');
  wsProd.setTabColor('#555555');

  ss.setActiveSheet(wsDash);
  SpreadsheetApp.flush();
}

function buildDashboard(ss, existingSheet) {
  const ws = existingSheet;
  ws.setName(TABS.dashboard);
  ws.setHiddenGridlines(true);
  ws.clearContents().clearFormats();

  ws.setRowHeight(1, 50);
  ws.getRange('A1:I1').merge()
    .setValue('SKYLINE MEDICAL ??Stock Management')
    .setBackground('#1a1a2e')
    .setFontFamily('Arial').setFontSize(16).setFontWeight('bold').setFontColor('#FFFFFF')
    .setHorizontalAlignment('center').setVerticalAlignment('middle');

  ws.getRange('A2:I2').merge()
    .setValue('Auto-updated via Apps Script')
    .setFontFamily('Arial').setFontSize(10).setFontColor('#888888')
    .setHorizontalAlignment('center');
  ws.setRowHeight(2, 18);

  [4, 5, 6, 7, 8].forEach(r => ws.setRowHeight(r, 22));
  ws.setRowHeight(6, 36);

  const cards = [
    {
      col:'B',
      label:'CPAP/BPAP Devices',
      formula:'=COUNTIF(Stock_Device!E:E,"Active")',
      bg:'#dce8f7',
      fg:'#1a4fa0'
    },
    {
      col:'D',
      label:'Masks',
      formula:'=SUMIF(Stock_Mask!E:E,"Active",Stock_Mask!M:M)',
      bg:'#d6f0e3',
      fg:'#1a7a4a'
    },
    {
      col:'F',
      label:'Accessories',
      formula:'=SUMIF(Stock_Accessory!E:E,"Active",Stock_Accessory!J:J)',
      bg:'#fdf3d0',
      fg:'#a06a00'
    },
    {
      col:'H',
      label:'Total In Stock',
      formula:'=COUNTIF(Stock_Device!E:E,"Active")+SUMIF(Stock_Mask!E:E,"Active",Stock_Mask!M:M)+SUMIF(Stock_Accessory!E:E,"Active",Stock_Accessory!J:J)',
      bg:'#ece5f7',
      fg:'#4a3580'
    }
  ];

  cards.forEach(c => {
    ws.getRange(c.col+'4').setValue(c.label)
      .setBackground(c.bg).setFontFamily('Arial').setFontSize(10).setFontWeight('bold').setFontColor(c.fg)
      .setHorizontalAlignment('center').setVerticalAlignment('middle')
      .setBorder(true,true,false,true,false,false,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID_MEDIUM);

    ws.getRange(c.col+'5').setBackground(c.bg)
      .setBorder(false,true,false,true,false,false,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID_MEDIUM);

    ws.getRange(c.col+'6').setFormula(c.formula)
      .setBackground(c.bg).setFontFamily('Arial').setFontSize(24).setFontWeight('bold').setFontColor(c.fg)
      .setHorizontalAlignment('center').setVerticalAlignment('middle')
      .setBorder(false,true,false,true,false,false,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID_MEDIUM);

    ws.getRange(c.col+'7').setBackground(c.bg)
      .setBorder(false,true,false,true,false,false,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID_MEDIUM);

    ws.getRange(c.col+'8').setValue('items in stock')
      .setBackground(c.bg).setFontFamily('Arial').setFontSize(9).setFontColor('#888888')
      .setHorizontalAlignment('center').setVerticalAlignment('middle')
      .setBorder(false,true,true,true,false,false,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID_MEDIUM);
  });

  const colW = {A:2,B:28,C:2,D:28,E:2,F:28,G:2,H:28,I:2};
  Object.keys(colW).forEach(c => {
    ws.setColumnWidth(ws.getRange(c+'1').getColumn(), colW[c] * 7);
  });

  buildDashboardSummaryTables(ws);

  return ws;
}

function buildDashboardSummaryTables(ws) {
  const sections = [
    {
      title: 'Device Stock by Model',
      titleRange: 'A11:D11',
      headerRange: 'A12:D12',
      formulaCell: 'A12',
      color: '#1a4fa0',
      formula: "=QUERY(Stock_Device!D3:G,\"select G,D,F,count(D) where E = 'Active' group by G,D,F label G 'Model', D 'Product Name', F 'Brand', count(D) 'Qty'\",0)"
    },
    {
      title: 'Mask Stock by Model / Size',
      titleRange: 'F11:K11',
      headerRange: 'F12:K12',
      formulaCell: 'F12',
      color: '#1a7a4a',
      formula: "=QUERY(Stock_Mask!D3:M,\"select G,H,I,D,F,sum(M) where E = 'Active' group by G,H,I,D,F label G 'Model', H 'Size', I 'REF', D 'Product Name', F 'Brand', sum(M) 'Qty'\",0)"
    },
    {
      title: 'Accessory Stock by Product',
      titleRange: 'A30:D30',
      headerRange: 'A31:D31',
      formulaCell: 'A31',
      color: '#a06a00',
      formula: "=QUERY(Stock_Accessory!D3:J,\"select D,F,sum(J) where E = 'Active' group by D,F label D 'Product Name', F 'Brand', sum(J) 'Qty'\",0)"
    },
  ];

  ws.getRange('A10:K48').clearContent().clearFormat();

  sections.forEach(s => {
    ws.getRange(s.titleRange).merge()
      .setValue(s.title)
      .setBackground(s.color)
      .setFontFamily('Arial').setFontSize(11).setFontWeight('bold').setFontColor('#FFFFFF')
      .setHorizontalAlignment('center').setVerticalAlignment('middle');

    ws.getRange(s.formulaCell).setFormula(s.formula);
    ws.getRange(s.headerRange)
      .setBackground('#f0f2f5')
      .setFontFamily('Arial').setFontSize(9).setFontWeight('bold').setFontColor('#333333')
      .setHorizontalAlignment('center')
      .setBorder(true,true,true,true,true,true,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID);
  });

  ws.getRange('A12:K48')
    .setFontFamily('Arial')
    .setFontSize(9)
    .setVerticalAlignment('middle');
  ws.getRange('D12:D48').setHorizontalAlignment('center');
  ws.getRange('K12:K48').setHorizontalAlignment('center');
  ws.getRange('D31:D48').setHorizontalAlignment('center');

  [11, 30].forEach(r => ws.setRowHeight(r, 24));
  ws.setColumnWidth(1, 130);  // A
  ws.setColumnWidth(2, 210);  // B
  ws.setColumnWidth(3, 130);  // C
  ws.setColumnWidth(4, 60);   // D
  ws.setColumnWidth(6, 120);  // F
  ws.setColumnWidth(7, 55);   // G
  ws.setColumnWidth(8, 80);   // H
  ws.setColumnWidth(9, 210);  // I
  ws.setColumnWidth(10, 130); // J
  ws.setColumnWidth(11, 60);  // K
}

function rebuildDashboard() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let ws = ss.getSheetByName(TABS.dashboard);
  if (!ws) ws = ss.insertSheet(TABS.dashboard, 0);
  buildDashboard(ss, ws);
  ss.setActiveSheet(ws);
}

function syncProductDB() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(TABS.productdb);
  if (!sheet) sheet = ss.insertSheet(TABS.productdb);

  sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), PRODUCTDB_HEADERS.length)).breakApart();
  sheet.getRange(1, 1, 1, PRODUCTDB_HEADERS.length).merge().setValue('Product Database - Skyline Medical')
    .setBackground('#333333').setFontFamily('Arial').setFontSize(13).setFontWeight('bold').setFontColor('#FFFFFF')
    .setHorizontalAlignment('center').setVerticalAlignment('middle');
  styleHeaderRow(sheet, 2, PRODUCTDB_HEADERS, '#333333');

  const existingRows = sheet.getLastRow() >= 3
    ? sheet.getRange(3, 1, sheet.getLastRow() - 2, Math.max(sheet.getLastColumn(), PRODUCTDB_HEADERS.length)).getValues()
    : [];
  const legacyRows = existingRows.filter(r => String(r[4] || '').trim()).map(r => normalizeProductDBRow_(r));
  const seedRows = getLegacyProductDBSeed_().map(r => normalizeProductDBRow_(r));
  const masterRows = SKYLINE_PRODUCT_MASTER.map(r => normalizeProductDBRow_(r));
  const combined = dedupeProductRows_(legacyRows.concat(seedRows).concat(masterRows));

  if (sheet.getLastRow() > 2) {
    sheet.getRange(3, 1, sheet.getLastRow() - 2, Math.max(sheet.getLastColumn(), PRODUCTDB_HEADERS.length)).clearContent();
  }
  if (combined.length) {
    sheet.getRange(3, 1, combined.length, PRODUCTDB_HEADERS.length).setValues(combined);
    combined.forEach((_, i) => {
      const r = i + 3;
      sheet.getRange(r, 1, 1, PRODUCTDB_HEADERS.length)
        .setBackground(r % 2 === 0 ? '#f7f8fa' : '#FFFFFF')
        .setFontFamily('Arial').setFontSize(10)
        .setBorder(true,true,true,true,true,true,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID);
    });
  }
  SpreadsheetApp.flush();
  Logger.log('ProductDB synced: ' + combined.length + ' rows (' + masterRows.length + ' master SKU)');
}

function normalizeProductDBRow_(r) {
  const row = (r || []).slice(0, PRODUCTDB_HEADERS.length);
  while (row.length < PRODUCTDB_HEADERS.length) row.push('');
  row[6] = normalizeType_(row[6], row[11], row[4]);
  if (!String(row[10] || '').trim()) row[10] = String(row[9] || row[0] || row[4] || '').trim();
  row[12] = toBool_(row[12]) || hasBarcodePattern_(row[0]) || hasBarcodePattern_(row[1]) || hasBarcodePattern_(row[2]) || hasBarcodePattern_(row[3]);
  if (!String(row[13] || '').trim()) row[13] = String(row[4] || '').trim();
  if (!String(row[14] || '').trim()) row[14] = String(row[4] || '').trim();
  return row;
}

function dedupeProductRows_(rows) {
  const out = [];
  const seen = new Set();
  rows.forEach(r => {
    if (!String(r[4] || '').trim()) return;
    const scanKey = [r[0], r[1], r[2], r[3]].join('|').toUpperCase();
    const manualKey = ['MANUAL', r[10], r[4], r[5], r[8]].join('|').toUpperCase();
    const key = scanKey.replace(/\|/g, '') ? scanKey : manualKey;
    if (seen.has(key)) return;
    seen.add(key);
    out.push(r);
  });
  return out;
}

function learnProductMapping_(ss, b) {
  const m = b && b.learnMapping ? b.learnMapping : null;
  if (!m) return false;

  const productCode = String(m.product_code || m.productCode || b.productCode || '').trim();
  const gtin = String(m.gtin || '').replace(/[^\x20-\x7E]/g, '').replace(/\s/g, '').replace(/^0+/, '').toUpperCase();
  const sn = String(m.sn || '').replace(/[^\x20-\x7E]/g, '').replace(/\s/g, '').toUpperCase();
  const lot = normalizeLot(String(m.lot || ''));
  if (!productCode || (!gtin && !sn && !lot)) return false;

  const sheet = buildProductDBSheet(ss);
  const last = sheet.getLastRow();
  const width = Math.max(sheet.getLastColumn(), PRODUCTDB_HEADERS.length);
  const rows = last >= 3
    ? sheet.getRange(3, 1, last - 2, width).getValues().map(r => normalizeProductDBRow_(r))
    : [];

  const source = rows.find(r => String(r[10] || '').trim() === productCode);
  if (!source) {
    Logger.log('Learn ProductDB mapping skipped: product_code not found: ' + productCode);
    return false;
  }

  const learned = normalizeProductDBRow_([
    gtin,
    sn,
    '',
    lot,
    source[4],
    source[5],
    source[6],
    source[7],
    source[8],
    source[9],
    source[10],
    source[11],
    true
  ]);

  const combined = dedupeProductRows_(rows.concat([learned]));
  if (combined.length === rows.length) return false;

  if (last > 2) {
    sheet.getRange(3, 1, last - 2, width).clearContent();
  }
  sheet.getRange(3, 1, combined.length, PRODUCTDB_HEADERS.length).setValues(combined);
  combined.forEach((_, i) => {
    const r = i + 3;
    sheet.getRange(r, 1, 1, PRODUCTDB_HEADERS.length)
      .setBackground(r % 2 === 0 ? '#f7f8fa' : '#FFFFFF')
      .setFontFamily('Arial').setFontSize(10)
      .setBorder(true,true,true,true,true,true,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID);
  });
  SpreadsheetApp.flush();
  Logger.log('Learn ProductDB mapping added for ' + productCode);
  return true;
}

function ocrLabel_(base64Image) {
  let fileId = '';
  try {
    const parsed = parseBase64Image_(base64Image);
    if (!parsed.bytes.length) return { text: '' };
    const blob = Utilities.newBlob(parsed.bytes, parsed.mimeType, 'ocr_label_' + Date.now());
    const file = Drive.Files.insert(
      { title: 'ocr_tmp_' + Date.now(), mimeType: blob.getContentType() },
      blob,
      { convert: true }
    );
    fileId = file.id;
    const doc = DocumentApp.openById(fileId);
    const text = doc.getBody().getText();
    DriveApp.getFileById(fileId).setTrashed(true);
    return { text: text || '' };
  } catch(ex) {
    Logger.log('ocr_label Drive OCR failed: ' + (ex.message || ex.toString()));
    if (fileId) {
      try { DriveApp.getFileById(fileId).setTrashed(true); } catch(cleanupErr) {}
    }
    return { text: '' };
  }
}

function uploadProductPhoto_(base64Image, filename) {
  try {
    const parsed = parseBase64Image_(base64Image);
    if (!parsed.bytes.length) throw new Error('Missing image');
    const safeName = sanitizeFilename_(filename || ('product-' + Utilities.formatDate(new Date(), TZ, 'yyyyMMdd-HHmmss') + '.jpg'));
    const folder = getOrCreateDriveFolder_('Product Photos');
    const blob = Utilities.newBlob(parsed.bytes, parsed.mimeType, safeName);
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return { url: 'https://drive.google.com/uc?id=' + file.getId() };
  } catch(ex) {
    Logger.log('upload_product_photo failed: ' + (ex.message || ex.toString()));
    throw ex;
  }
}

function addProduct_(ss, b) {
  const productName = String(b.productName || b.name || '').trim();
  if (!productName) throw new Error('\u0e01\u0e23\u0e38\u0e13\u0e32\u0e01\u0e23\u0e2d\u0e01\u0e0a\u0e37\u0e48\u0e2d\u0e2a\u0e34\u0e19\u0e04\u0e49\u0e32');

  const normalizedGtin = normalizeGtinForCompare_(gtin);
  const productType = normalizeType_(String(b.type || '').trim(), String(b.category || '').trim(), productName);
  const productModel = String(b.model || '').trim();
  const productSize = String(b.size || '').trim();
  const productRef = String(b.ref || '').trim();
  if (normalizedGtin && readProductDB(ss).some(p => isDuplicateProductForAdd_(p, {
    gtin: gtin,
    name: productName,
    type: productType,
    model: productModel,
    size: productSize,

  }))) {

  }

  const sheet = getOrCreateAddProductSheet_(ss);
  const idx = ensureProductDBColumns_(sheet);
  const productCode = String(b.productCode || b.product_code || b.code || '').trim() || makeProductCode_(b, productName, gtin);
  const row = new Array(PRODUCTDB_HEADERS.length).fill('');
  const set = (name, value) => {
    const col = idx[name];
    if (col != null) row[col - 1] = value;
  };

  set('GTIN', gtin);
  set('Product Name', productName);
  set('Brand', String(b.brand || '').trim());
  set('Type', productType);
  set('Model', productModel);
  set('Size', productSize);
  set('REF', productRef);
  set('Product Code', productCode);
  set('Category', String(b.category || '').trim());
  set('has_barcode', toBool_(b.hasBarcode) || !!gtin);
  set('Stock Display Name', String(b.stockName || '').trim() || productName);

  set('Photo URL', String(b.photoUrl || '').trim());
  set('Added Date', new Date().toISOString());

  sheet.appendRow(row);
  const newRow = sheet.getLastRow();
  sheet.getRange(newRow, 1, 1, PRODUCTDB_HEADERS.length)
    .setBackground(newRow % 2 === 0 ? '#f7f8fa' : '#FFFFFF')
    .setFontFamily('Arial').setFontSize(10)
    .setBorder(true,true,true,true,true,true,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID);
  return {productCode: productCode};
}

function normalizeGtinForCompare_(value) {
  return String(value || '').replace(/[^\x20-\x7E]/g, '').replace(/\s/g, '').replace(/^0+/, '').trim();
}

function normalizeProductIdentityText_(value) {
  return String(value || '').trim().toUpperCase();
}

function sameProductIdentityText_(a, b) {
  return normalizeProductIdentityText_(a) === normalizeProductIdentityText_(b);
}

function isMaskProduct_(p) {
  return normalizeType_(p.type, p.category, p.product_name || p.name) === 'mask';
}

function isDuplicateProductForAdd_(existing, candidate) {
  if (normalizeGtinForCompare_(existing && existing.gtin) !== normalizeGtinForCompare_(candidate && candidate.gtin)) return false;
  if (!isMaskProduct_(existing) && !isMaskProduct_(candidate)) return true;

  const sameRef = normalizeProductIdentityText_(existing.ref) &&
    sameProductIdentityText_(existing.ref, candidate.ref);
  const sameModelSize = sameProductIdentityText_(existing.model, candidate.model) &&
    sameProductIdentityText_(existing.size, candidate.size);
  const existingName = existing.product_name || existing.name || '';
  const sameNameSize = sameProductIdentityText_(existingName, candidate.name) &&
    sameProductIdentityText_(existing.size, candidate.size);

  return !!(sameRef || sameModelSize || sameNameSize);
}

function stripBase64Prefix_(value) {
  return String(value || '').replace(/^data:[^;]+;base64,/, '').trim();
}

function parseBase64Image_(value) {
  const raw = String(value || '');
  const m = raw.match(/^data:([^;]+);base64,(.+)$/);
  const mimeType = m ? m[1] : 'image/jpeg';
  const data = m ? m[2] : raw;
  return { mimeType: mimeType, bytes: Utilities.base64Decode(data) };
}

function getOrCreateDriveFolder_(name) {
  const it = DriveApp.getFoldersByName(name);
  return it.hasNext() ? it.next() : DriveApp.createFolder(name);
}

function sanitizeFilename_(name) {
  const s = String(name || '').replace(/[\\/:*?"<>|#%{}~&]/g, '-').trim();
  return s || 'product-photo.jpg';
}

function normalizeType_(type, category, name) {
  const t = String(type || '').trim().toLowerCase();
  if (['device','mask','accessory'].indexOf(t) !== -1) return t;
  const c = String(category || '');
  const n = String(name || '').toLowerCase();
  const m = c.match(/^\s*(\d+)/);
  const catNo = m ? parseInt(m[1], 10) : 0;
  if ([5,6,7].indexOf(catNo) !== -1) return 'mask';
  if (/filter|hose|circuit|cushion|humidifier|chamber|accessor|chin strap/i.test(c + ' ' + n)) return 'accessory';
  if ([1,2,3,4].indexOf(catNo) !== -1) return 'device';
  return 'accessory';
}

function toBool_(v) {
  if (v === true) return true;
  const s = String(v || '').trim().toUpperCase();
  return s === 'TRUE' || s === 'YES' || s === '1';
}

function hasBarcodePattern_(v) {
  const s = String(v || '').trim().toUpperCase();
  if (!s || s === '-') return false;
  return /^\d{8,14}$/.test(s) || /(BY-|R30F|R25V|R20I|R20A|NM\d|PMI|HP\d|SPO2|\*)/.test(s);
}

function readProductDB(ss) {
  const out = [];
  const byGtin = {};

  getLegacyProductDBSeed_().concat(SKYLINE_PRODUCT_MASTER).forEach(r => {
    const p = productFromMasterRow_(normalizeProductDBRow_(r));
    addMergedProduct_(out, byGtin, p, false);
  });

  const sheet = ss.getSheetByName(TABS.productdb);
  if (!sheet) return out;

  const last = sheet.getLastRow();
  const width = Math.max(sheet.getLastColumn(), PRODUCTDB_HEADERS.length, 12);
  if (last < 1) return out;

  const values = sheet.getRange(1, 1, last, width).getValues();
  const headerRows = [];
  values.forEach((row, i) => {
    const joined = row.map(v => String(v || '').trim()).join('|');
    if (joined.indexOf('GTIN') !== -1 && joined.indexOf('Product Name') !== -1) {
      headerRows.push({ index: i, headers: row.map(h => String(h || '').trim()) });
    }
  });

  headerRows.forEach(hr => {
    const idx = {};
    hr.headers.forEach((h, i) => {
      if (h) idx[h] = i;
    });
    for (let i = hr.index + 1; i < values.length; i++) {
      const row = values[i];
      if (isHeaderLikeProductRow_(row)) continue;
      const p = productFromSheetRow_(row, idx);
      addMergedProduct_(out, byGtin, p, true);
    }
  });

  return out;
}

function productFromSheetRow_(row, idx) {
  const hasFullSchema = idx['SN_Exact'] != null || idx['Product Code'] != null || idx['Search Aliases'] != null;
  const get = name => idx[name] == null ? '' : row[idx[name]];
  const looksLikeSimpleAppend = hasFullSchema &&
    String(row[1] || '').trim() &&
    ['device','mask','accessory'].indexOf(String(row[3] || '').trim().toLowerCase()) !== -1;
  if (looksLikeSimpleAppend) {
    return productFromSimpleProductRow_(row, {
      GTIN: 0,
      'Product Name': 1,
      Brand: 2,
      Type: 3,
      Model: 4,
      Size: 5,
      REF: 6,
      Category: 7,
      Aliases: 8,
      has_barcode: 9,
      'Photo URL': 10,
      'Added Date': 11
    });
  }
  if (hasFullSchema) {
    const baseName = String(get('Product Name') || '').trim();
    const stockName = String(get('Stock Display Name') || '').trim();
    const poName = String(get('PO Print Name') || '').trim();
    const aliases = String(get('Search Aliases') || get('Aliases') || '').trim();

      gtin:         String(get('GTIN') || '').trim(),
      sn_exact:     String(get('SN_Exact') || '').trim().toUpperCase(),
      sn_pfx:       String(get('SN_Prefix') || '').trim().toUpperCase(),
      lot_pfx:      String(get('LOT_Prefix') || '').trim().toUpperCase(),
      name:         stockName || baseName,
      product_name: baseName,
      stock_name:   stockName,
      po_name:      poName,
      aliases:      aliases,
      brand:        String(get('Brand') || '').trim(),
      type:         normalizeType_(get('Type'), get('Category'), baseName),
      model:        String(get('Model') || '').trim(),
      size:         String(get('Size') || '').trim(),
      ref:          String(get('REF') || '').trim(),

      category:     String(get('Category') || '').trim(),

      photoUrl:     String(get('Photo URL') || '').trim(),
      addedDate:    String(get('Added Date') || '').trim()
    };
  }

  return productFromSimpleProductRow_(row, idx);
}

function addMergedProduct_(out, byGtin, p, preferExistingGtin) {
  if (!p || !String(p.product_name || p.name || '').trim()) return;
  const key = normalizeGtinForCompare_(p.gtin);
  const identity = productMergeIdentity_(p);
  const mergeKey = key ? key + '|' + identity : 'NO_GTIN|' + identity;
  if (byGtin[mergeKey] != null) {
    if (preferExistingGtin) out[byGtin[mergeKey]] = p;
    return;
  }
  byGtin[mergeKey] = out.length;
  out.push(p);
}

function productMergeIdentity_(p) {
  const type = normalizeType_(p.type, p.category, p.product_name || p.name);
  const productCode = normalizeProductIdentityText_(p.product_code);
  if (productCode) return [type, 'CODE', productCode].join('|');
  if (type === 'mask') {
    return [
      type,
      normalizeProductIdentityText_(p.ref),
      normalizeProductIdentityText_(p.model),
      normalizeProductIdentityText_(p.size),
      normalizeProductIdentityText_(p.product_name || p.name)
    ].join('|');
  }
  return [
    type,
    normalizeProductIdentityText_(p.ref),
    normalizeProductIdentityText_(p.product_name || p.name)
  ].join('|');
}

function isHeaderLikeProductRow_(row) {
  return String(row[0] || '').trim() === 'GTIN' || String(row[4] || '').trim() === 'Product Name';
}

function ensureProductDBColumns_(sheet) {
  if (!sheet) throw new Error('ProductDB sheet not found');
  const width = Math.max(sheet.getLastColumn(), PRODUCTDB_HEADERS.length);
  const headers = sheet.getRange(2, 1, 1, width).getValues()[0].map(h => String(h || '').trim());
  let changed = false;

  PRODUCTDB_HEADERS.forEach((h, i) => {
    if (headers[i] !== h) {
      headers[i] = h;
      changed = true;
    }
  });

  if (changed) {
    sheet.getRange(2, 1, 1, PRODUCTDB_HEADERS.length).setValues([headers.slice(0, PRODUCTDB_HEADERS.length)]);
    styleHeaderRow(sheet, 2, PRODUCTDB_HEADERS, '#333333');
  }
  return PRODUCTDB_HEADERS.reduce((idx, h, i) => {
    idx[h] = i + 1;
    return idx;
  }, {});
}

function updateProductNames_(ss, b) {
  const code = normalizeText(b.productCode || b.product_code);
  if (!code) throw new Error('Missing product code');

  const sheet = buildProductDBSheet(ss);
  const cols = ensureProductDBColumns_(sheet);
  const last = sheet.getLastRow();
  if (last < DATA_START) throw new Error('ProductDB is empty');

  const codeVals = sheet.getRange(DATA_START, cols['Product Code'], last - DATA_START + 1, 1).getValues();
  let row = -1;
  for (let i = 0; i < codeVals.length; i++) {
    if (normalizeText(codeVals[i][0]).toUpperCase() === code.toUpperCase()) {
      row = i + DATA_START;
      break;
    }
  }
  if (row === -1) throw new Error('Product code not found: ' + code);

  const stockName = normalizeText(b.stockName || b.stock_name).slice(0, 160);
  const poName = normalizeText(b.poName || b.po_name).slice(0, 220);
  const aliases = normalizeText(b.aliases).slice(0, 500);

  sheet.getRange(row, cols['Stock Display Name']).setValue(stockName);
  sheet.getRange(row, cols['PO Print Name']).setValue(poName);
  sheet.getRange(row, cols['Search Aliases']).setValue(aliases);

  const note = [
    'Name edit',
    'code=' + code,
    'stock=' + stockName,
    'po=' + poName,
    'aliases=' + aliases,
    'by=' + (b.operator || b.staffName || b.user || '')
  ].join(' | ');

  appendHistory(ss, {
    type: 'accessory',
    name: normalizeText(sheet.getRange(row, cols['Product Name']).getValue()) || code,
    brand: normalizeText(sheet.getRange(row, cols['Brand']).getValue()),
    productCode: code,
    qty: 1,
    purpose: 'product update',
    recipient: '',
    user: b.user,
    operator: b.operator,
    note: note
  }, Utilities.formatDate(new Date(), TZ, 'dd/MM/yyyy'), Utilities.formatDate(new Date(), TZ, 'HH:mm'), 'Product Update');

  SpreadsheetApp.flush();
  return { row: row, productCode: code, stockName: stockName, poName: poName, aliases: aliases };
}

function backfillDeviceSheet_(sheet, products) {
  if (!sheet || sheet.getLastRow() < DATA_START) return 0;
  const values = sheet.getRange(DATA_START, 1, sheet.getLastRow() - DATA_START + 1, sheet.getLastColumn()).getValues();
  let updated = 0;
  values.forEach((row, i) => {
    const p = lookupProductForBackfill_(products, {
      type:'device',
      gtin: row[9],
      lot: row[DEV_COL_LOT - 1],
      sn: row[DEV_COL_SN - 1]
    });
    if (!p || p.type !== 'device') return;
    const r = DATA_START + i;
    updated += setIfChanged_(sheet, r, 4, p.name);
    updated += setIfChanged_(sheet, r, 6, p.brand);
    updated += setIfChanged_(sheet, r, 7, p.model);
  });
  return updated;
}

function setIfChanged_(sheet, row, col, value) {
  value = String(value || '').trim();
  if (!value) return 0;
  const cur = String(sheet.getRange(row, col).getValue() || '').trim();
  if (cur === value) return 0;
  sheet.getRange(row, col).setValue(value);
  return 1;
}

function lookupProductForBackfill_(products, d) {
  const gtin = String(d.gtin || '').replace(/\s/g, '').replace(/^0+/, '').toUpperCase();
  const lot = normalizeLot(d.lot);
  const sn = String(d.sn || '').replace(/\s/g, '').toUpperCase();
  const refTokens = extractRefTokens_([d.ref, d.productCode, d.raw, d.lot].join(' '));
  const codeToken = normalizeSkuToken_(String(d.productCode || d.raw || ''));
  const inferred = inferProductFromLot_(lot || d.raw || '');
  const refCandidates = [];
  const codeCandidates = [];
  const wildcardLotHits = [];

  for (const p of products || []) {
    if (p.type && d.type && p.type !== d.type) continue;
    const pGtin = String(p.gtin || '').replace(/^0+/, '').toUpperCase();
    const pRef = normalizeRefToken_(p.ref || '');
    const pCode = normalizeSkuToken_(p.product_code || '');
    if (pGtin && gtin && pGtin === gtin) return mergeLotDetails_(p, lot, {preferProductIdentity:true});
    if (p.sn_exact && sn && p.sn_exact === sn) return mergeLotDetails_(p, lot);
    if (p.sn_pfx && sn && sn.startsWith(p.sn_pfx)) return mergeLotDetails_(p, lot);
    if (pRef && refTokens.indexOf(pRef) !== -1) refCandidates.push(p);
    if (pCode && pCode !== '-' && codeToken === pCode) codeCandidates.push(p);
    if (p.lot_pfx && lot && matchProductPattern_(lot, p.lot_pfx)) {
      if (String(p.lot_pfx).indexOf('*') !== -1) {
        wildcardLotHits.push(p);
      } else {
        return mergeLotDetails_(p, lot);
      }
    }
  }

  const refHit = pickUniqueProductCandidate_(refCandidates, inferred);
  if (refHit) return mergeLotDetails_(refHit, lot || d.raw || '');
  const codeHit = pickUniqueProductCandidate_(codeCandidates, inferred);
  if (codeHit) return mergeLotDetails_(codeHit, lot || d.raw || '');
  const wildcardLotHit = pickWildcardLotHit_(wildcardLotHits, inferred);
  if (wildcardLotHit) return mergeLotDetails_(wildcardLotHit, lot);
  return inferred;
}

function pickUniqueProductCandidate_(rows, hint) {
  rows = rows || [];
  if (rows.length === 1) return rows[0];
  if (!rows.length || !hint) return null;
  let narrowed = rows;
  if (hint.model) narrowed = narrowed.filter(p => normalizeText(p.model).toUpperCase() === normalizeText(hint.model).toUpperCase());
  if (hint.size) narrowed = narrowed.filter(p => normalizeText(p.size).toUpperCase() === normalizeText(hint.size).toUpperCase());
  return narrowed.length === 1 ? narrowed[0] : null;
}

function pickWildcardLotHit_(rows, hint) {
  const hinted = pickUniqueProductCandidate_(rows || [], hint);
  if (hinted) return hinted;
  return (rows || []).slice().sort(function(a, b) {
    return wildcardPatternSpecificity_(b.lot_pfx) - wildcardPatternSpecificity_(a.lot_pfx);
  })[0] || null;
}

function wildcardPatternSpecificity_(pattern) {
  return normalizeSkuToken_(pattern).length;
}

function matchProductPattern_(value, pattern) {
  const v = String(value || '').trim().toUpperCase();
  const p = String(pattern || '').trim().toUpperCase();
  if (!v || !p) return false;
  if (p.indexOf('*') === -1) return v.indexOf(p) === 0;
  const re = new RegExp('^' + p.split('*').map(escapeRegExp_).join('.*') + '$');
  return re.test(v);
}

function escapeRegExp_(s) {
  return String(s || '').replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

function mergeLotDetails_(product, lot, opts) {
  const inferred = inferProductFromLot_(lot) || {};
  opts = opts || {};
  const merged = Object.assign({}, inferred || {});
  Object.keys(product || {}).forEach(k => {
    const v = product[k];
    if (v !== undefined && v !== null && v !== '') merged[k] = v;
  });
  if (opts.preferProductIdentity) {
    if (!merged.size && inferred.size) merged.size = inferred.size;
    if (!merged.ref && inferred.ref) merged.ref = inferred.ref;
    return merged;
  }
  if (inferred.size) merged.size = inferred.size;
  if (inferred.ref) merged.ref = inferred.ref;
  return merged;
}

function enrichTransactionProduct_(ss, b) {
  if (!b || !b.type) return b;
  const products = readProductDB(ss);
  const hit = lookupProductForBackfill_(products, {
    type: b.type,
    gtin: b.gtin,
    lot: b.lot,
    sn: b.sn,
    ref: b.ref,
    productCode: b.productCode,
    raw: [b.name, b.ref, b.productCode, b.lot].join(' ')
  }) || inferProductFromLot_([b.lot, b.ref, b.productCode, b.name].join(' '));

  if (!hit) return b;
  b.name = String(b.name || hit.name || '').trim();
  b.brand = String(b.brand || hit.brand || '').trim();
  b.model = String(b.model || hit.model || '').trim();
  b.size = String(b.size || hit.size || '').trim();
  b.ref = String(b.ref || hit.ref || '').trim();
  return b;
}

function transactionIdentityError_(b, action) {
  if (action !== 'in') return '';
  if (b.type === 'device' && !String(b.model || '').trim()) return 'Missing device model mapping';
  if (b.type === 'mask' && !isSizelessMaskProduct_(b) && (!String(b.model || '').trim() || !String(b.size || '').trim())) {
    return 'Missing mask model/size mapping';
  }
  return '';
}

function isSizelessMaskProduct_(p) {
  const text = normalizeSkuToken_([p && p.name, p && p.model, p && p.ref, p && p.productCode].join(' '));
  return text.indexOf('EASEFITPMI') !== -1 || text.indexOf('PILLOW') !== -1;
}

function normalizeSkuToken_(v) {
  return String(v || '').toUpperCase().replace(/[^A-Z0-9]/g, '');
}

function normalizeIncomingGtin_(raw) {
  raw = String(raw || '').replace(/[^\x20-\x7E]/g, '').replace(/\s/g, '').trim();
  if (!raw) return '';
  let m = raw.match(/\(01\)(\d{8,14})/);
  if (m) return m[1].replace(/^0+/, '');
  m = raw.match(/^01(\d{14})/);
  if (m) return m[1].replace(/^0+/, '');
  m = raw.match(/\b(\d{8,14})\b/);
  if (m) return m[1].replace(/^0+/, '');
  return '';
}

function normalizeRefToken_(v) {
  return String(v || '').toUpperCase().replace(/^REF/i, '').replace(/[^A-Z0-9]/g, '');
}

function extractRefTokens_(text) {
  const raw = String(text || '').toUpperCase();
  const out = [];
  const add = v => {
    v = normalizeRefToken_(v);
    if (v && out.indexOf(v) === -1) out.push(v);
  };
  let m;
  const refRe = /\bREF\s*[:#-]?\s*([A-Z0-9._/-]{3,})/g;
  while ((m = refRe.exec(raw)) !== null) add(m[1]);
  const looseRe = /\b(PT?\d{4,6}|\d{5})\b/g;
  while ((m = looseRe.exec(raw)) !== null) add(m[1]);
  return out;
}

function inferProductFromLot_(lot) {
  lot = normalizeSkuToken_(normalizeLot(lot));
  if (!lot) return null;
  const rules = [
    { test: l => l.indexOf('NM6A') !== -1, type:'mask', model:'EaseFit NM6A', name:'Beyond EaseFit NM6A', brand:'Hunan Beyond' },
    { test: l => l.indexOf('NM6') !== -1,  type:'mask', model:'EaseFit NM6',  name:'Beyond EaseFit NM6',  brand:'Hunan Beyond' },
    { test: l => l.indexOf('NM5A') !== -1, type:'mask', model:'EaseFit NM5A', name:'Beyond EaseFit NM5A', brand:'Hunan Beyond' },
    { test: l => l.indexOf('NM5') !== -1,  type:'mask', model:'EaseFit NM5',  name:'Beyond EaseFit NM5',  brand:'Hunan Beyond' },
    { test: l => l.indexOf('PMI') !== -1,  type:'mask', model:'EaseFit PMI',  name:'Beyond EaseFit PMI',  brand:'Hunan Beyond' },
    { test: l => /HP\d*$/.test(l) || l.indexOf('HP1') !== -1, type:'accessory', model:'HP1', name:'Breathing Tube HP1', brand:'Hunan Beyond' },
    { test: l => l.indexOf('SPO2') !== -1, type:'accessory', model:'SpO2 Kit', name:'Beyond SpO2 Kits', brand:'Hunan Beyond' },
    { test: l => l.indexOf('R30F108') !== -1, type:'device', model:'R-30F ASV', name:'Beyond R-30F ASV', brand:'Hunan Beyond' },
    { test: l => l.indexOf('R30F107') !== -1 || l.indexOf('R30F') !== -1, type:'device', model:'R-30F', name:'Beyond R-30F BPAP', brand:'Hunan Beyond' },
    { test: l => l.indexOf('R25V') !== -1, type:'device', model:'R-25V', name:'Beyond R-25V BPAP', brand:'Hunan Beyond' },
    { test: l => l.indexOf('R20A') !== -1, type:'device', model:'R-20A', name:'Beyond R-20A CPAP', brand:'Hunan Beyond' },
    { test: l => l.indexOf('R20') !== -1, type:'device', model:'R-20i', name:'Beyond R-20i CPAP', brand:'Hunan Beyond' },
  ];
  const matched = rules.find(r => r.test(lot));
  const refRules = [
    { model:'EaseFit NM5A', size:'L', ref:'78072' },
    { model:'EaseFit NM5A', size:'M', ref:'PT78044' },
    { model:'EaseFit NM6A', size:'M', ref:'78091' },
    { model:'EaseFit NM6',  size:'L', ref:'78084' },
    { model:'EaseFit NM6',  size:'M', ref:'78085' },

    { model:'EaseFit NM5',  size:'L', ref:'78066' },
    { model:'EaseFit NM5',  size:'S', ref:'78066' },
  ];
  const refRuleFromText = extractRefTokens_(lot).map(ref => {
    const matches = refRules.filter(r => normalizeRefToken_(r.ref) === ref);
    return matches.length === 1 ? matches[0] : null;
  }).filter(Boolean)[0];
  if (!matched && !refRuleFromText) return null;
  const sizeRules = [
    { test: l => l.indexOf('NM5AL') !== -1 || l.indexOf('NM6AL') !== -1 || l.indexOf('NM6L') !== -1 || l.indexOf('NM5L') !== -1 || (/L$/.test(l) && !/LOT$/.test(l)), size:'L' },
    { test: l => l.indexOf('NM5AM') !== -1 || l.indexOf('NM6AM') !== -1 || l.indexOf('NM6M') !== -1 || l.indexOf('NM5M') !== -1 || /M$/.test(l), size:'M' },
    { test: l => l.indexOf('NM5AS') !== -1 || l.indexOf('NM6AS') !== -1 || l.indexOf('NM6S') !== -1 || l.indexOf('NM5S') !== -1 || /S$/.test(l), size:'S' },
  ];
  const sizeRule = sizeRules.find(r => r.test(lot));
  const model = matched ? matched.model : refRuleFromText.model;
  const size = sizeRule ? sizeRule.size : (refRuleFromText ? refRuleFromText.size : '');
  const refRule = refRules.find(r => r.model === model && r.size === size);
  const base = matched || { type:'mask', model:model, name:'Beyond ' + model, brand:'Hunan Beyond' };
  return Object.assign({}, base, { size:size, ref: refRule ? refRule.ref : (refRuleFromText ? refRuleFromText.ref : '') });
}

function doGet(e) {
  const p = (e && e.parameter) ? e.parameter : {};
  const action = p.action || '';

  if (action === 'login') {
  const user = p.user || '';
  const pass = p.pass || '';
  const staffName = String(p.staffName || '').trim();
  if (USERS[user] && USERS[user] === pass && staffName) {
    return ok({ message:'ok', user:user, role:getUserRole(user), staffName:staffName, token:makeAuthToken(user, staffName) });
  }
  return err('Invalid login or missing staff name');
  }
  if (action === 'ping') {
    return ok({message:'Skyline API ready', ts:new Date().toISOString()});
  }

  if (action === 'stock') {
    try {
      const auth = requireAuth(p);
      if (!auth.ok) return err(auth.error);
      const ss = SpreadsheetApp.getActiveSpreadsheet();
      return ok({
        device: readStock(ss, 'device'),
        mask: readStock(ss, 'mask'),
        accessory: readStock(ss, 'accessory')
      });
    } catch(ex) {
      return err(ex.toString());
    }
  }

  if (action === 'export_all_data') {
    try {
      const auth = requireAuth(p, 'admin');
      if (!auth.ok) return err(auth.error);
      const ss = SpreadsheetApp.getActiveSpreadsheet();

    } catch(ex) {
      return err(ex.toString());
    }
  }

  if (action === 'history') {
    try {
      const auth = requireAuth(p);
      if (!auth.ok) return err(auth.error);

      const offset = parseInt(p.offset || '0', 10);
      const ss = SpreadsheetApp.getActiveSpreadsheet();
      return ok({history: readHistory(ss, n, offset)});
    } catch(ex) {
      return err(ex.toString());
    }

  if (action === 'sales_detail_alerts') {

function doPost(e) {
  let lock = null;
  try {
    const body = JSON.parse(e.postData.contents);
    const ss   = SpreadsheetApp.getActiveSpreadsheet();
    const now  = new Date();
    const date = Utilities.formatDate(now, TZ, 'dd/MM/yyyy');
    const time = Utilities.formatDate(now, TZ, 'HH:mm');


    if (body.action === 'login') {
      const user = String(body.user || '');
      const pass = String(body.pass || '');
      const staffName = String(body.staffName || '').trim();
      if (USERS[user] && USERS[user] === pass && staffName) {
        return ok({ message:'ok', user:user, role:getUserRole(user), staffName:staffName, token:makeAuthToken(user, staffName) });
      }
      return err('Invalid login or missing staff name');
    }


    const VALID_TYPES = ['device', 'mask', 'accessory'];
    if (['in', 'out', 'check', 'out_by_id'].includes(body.action)) {
      if (!VALID_TYPES.includes(body.type)) {
        return err('Invalid product type: ' + String(body.type || ''));
      }
    }

    const auth = requireAuth(body);
    if (!auth.ok) return err(auth.error);
    applyAuthContext_(body, auth);
    if (requiresWriteLock_(body.action)) {
      lock = LockService.getScriptLock();
      lock.waitLock(20000);
    }

    if (body.action === 'ocr_label') {
      try {
        return ok(ocrLabel_(body.image || body.base64 || body.base64Image || ''));
      } catch(ex) {
        Logger.log('ocr_label error: ' + ex.toString());
        return ok({ text: '' });
      }
    }

    if (body.action === 'upload_product_photo') {
      try {
        return ok(uploadProductPhoto_(body.image || body.base64 || body.base64Image || '', body.filename || ''));
      } catch(ex) {
        Logger.log('upload_product_photo error: ' + ex.toString());
        return err('Upload product photo failed: ' + (ex.message || ex.toString()));
      }
    }

    if (body.action === 'add_product') {
      try {
        addProduct_(ss, body.product || body);
        return ok({});
      } catch(ex) {
        Logger.log('add_product error: ' + ex.toString());
        return err(ex.message || ex.toString());
      }
    }

    if (body.action === 'in') {
      try { enrichTransactionProduct_(ss, body); } catch(enrichErr) { Logger.log('Stock In enrich skipped: ' + (enrichErr.message || enrichErr.toString())); }
      try { validateTransaction_(body, 'in'); } catch(vErr) { return err(vErr.message); }
      const identityErr = transactionIdentityError_(body, 'in');
      if (identityErr) return err(identityErr);
      const duplicate = findDuplicate(ss, body.type, body.sn, body.lot, body.name, body.size || '');
      if (duplicate) return err('DUPLICATE', { duplicate: duplicate });

      let stockInResult;
      try {
        stockInResult = doStockIn(ss, body, date, time);
      } catch(stockErr) {
        Logger.log('Stock In error: ' + stockErr.toString());
        return err('Stock In failed: ' + (stockErr.message || stockErr.toString()));
      }
      try {
        learnProductMapping_(ss, body);
      } catch(learnErr) {
        Logger.log('Learn ProductDB mapping failed after Stock In: ' + (learnErr.message || learnErr.toString()));
      }

      return ok(Object.assign({message:'in'}, typeof stockInResult === 'object' ? stockInResult : {row: stockInResult}));
    }

    if (body.action === 'out') {
      try { validateTransaction_(body, 'out'); } catch(vErr) { return err(vErr.message); }
      let r;
      try {
        r = doStockOut(ss, body, date, time);
      } catch(stockErr) {
        Logger.log('Stock Out error: ' + stockErr.toString());
        return err('Stock Out failed: ' + (stockErr.message || stockErr.toString()));
      }
      if (r !== -1) {
        try {
          learnProductMapping_(ss, body);
        } catch(learnErr) {
          Logger.log('Learn ProductDB mapping failed after Stock Out: ' + (learnErr.message || learnErr.toString()));
        }
      }
      return r === -1 ? err('Item not found') : ok({message:'out', row:r});
    }

    if (body.action === 'check') {
      const exists = checkDuplicate(ss, body.type, body.sn, body.lot, body.name, body.size || '');
      return ok({ exists });
    }

    if (body.action === 'out_by_id') {
      try { validateTransaction_(body, 'out'); } catch(vErr) { return err(vErr.message); }
      try {
        doStockOutByRow(ss, body, date, time);
      } catch(ex) {
        Logger.log('out_by_id error: ' + ex.toString());
        return err('out_by_id failed: ' + (ex.message || ex.toString()));
      }
      return ok({message:'out_by_id'});
    }

    if (body.action === 'batch_stocktake') {
      try {
        const result = doBatchStocktake(ss, body.items || [], date, time, body);
        return ok({message:'batch_stocktake', updated:result.updated});
      } catch(ex) {
        Logger.log('batch_stocktake error: ' + ex.toString());
        return err('Batch stocktake failed: ' + (ex.message || ex.toString()));
      }
    }

    if (body.action === 'update_product_names') {
      try {
        const result = updateProductNames_(ss, body);
        return ok(Object.assign({message:'update_product_names'}, result));
      } catch(ex) {
        Logger.log('update_product_names error: ' + ex.toString());
        return err('Update product names failed: ' + (ex.message || ex.toString()));
      }
    }

    if (body.action === 'update_sales_details') {
      try {
        const result = updateSalesDetails_(ss, body);
        return ok(Object.assign({message:'update_sales_details'}, result));
      } catch(ex) {
        Logger.log('update_sales_details error: ' + ex.toString());
        return err('Update sales details failed: ' + (ex.message || ex.toString()));
      }
    }

    if (body.action === 'sort') {
      const admin = requireAuth(body, 'admin');
      if (!admin.ok) return err(admin.error);
      try {
        const sheet = ss.getSheetByName(TABS.device);
        if (!sheet) return err('Device Sheet not found');
        sortDeviceSheet(sheet);
        return ok({message:'sorted'});
      } catch(ex) {
        Logger.log('sort error: ' + ex.toString());
        return err('sort failed: ' + (ex.message || ex.toString()));
      }
    }

    return err('Unknown action');
  } catch(ex) {
    Logger.log('doPost fatal error: ' + ex.toString());
    return err('Server error: ' + (ex.message || ex.toString()));
  } finally {
    if (lock) {
      try { lock.releaseLock(); } catch(lockErr) {}
    }
  }
}

function readStock(ss, type) {
  const sheet = ss.getSheetByName(TABS[type]);
  if (!sheet) return [];
  const last = sheet.getLastRow();
  if (last < DATA_START) return [];

  const headers = sheet.getRange(HDR_ROW,1,1,sheet.getLastColumn()).getValues()[0];
  const data = sheet.getRange(DATA_START,1,last-DATA_START+1,sheet.getLastColumn()).getValues();

  return data.map((row, i) => {
    const o = {_sheetRow: i + DATA_START};
    headers.forEach((h, j) => {
      if (h) o[h] = row[j];
    });
    return o;
  }).filter(r => {
    if (!r['Product Name']) return false;
    if (normalizeText(r['Status']).toLowerCase() !== 'active') return false;
    if (type !== 'device' && Number(r['Qty'] || 0) <= 0) return false;
    return true;
  });
}

function readSheetObjects_(sheet) {
  if (!sheet) return [];
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  if (lastRow < HDR_ROW || lastCol < 1) return [];

  const headers = sheet.getRange(HDR_ROW, 1, 1, lastCol).getValues()[0]
    .map((h, i) => String(h || ('Column ' + (i + 1))).trim());
  if (lastRow < DATA_START) return [];

  return sheet.getRange(DATA_START, 1, lastRow - DATA_START + 1, lastCol).getValues()
    .map((row, idx) => {
      const o = {_sheetRow: idx + DATA_START};
      headers.forEach((h, i) => { if (h) o[h] = row[i]; });
      return o;
    })
    .filter(o => Object.keys(o).some(k => k !== '_sheetRow' && String(o[k] || '').trim() !== ''));
}

function readHistory(ss, n, offset) {
  const sheet = ss.getSheetByName(TABS.history);
  if (!sheet) return [];
  ensureHistoryColumns_(sheet);
  const last = sheet.getLastRow();
  if (last < DATA_START) return [];

  const headers = sheet.getRange(HDR_ROW, 1, 1, sheet.getLastColumn()).getValues()[0];
  const totalRows = last - DATA_START + 1;


  const off = parseInt(offset || 0, 10);
  if (off >= totalRows) return [];
  const startFromBottom = Math.max(DATA_START, last - off - n + 1);
  const endRow = Math.max(DATA_START, last - off);
  const count = endRow - startFromBottom + 1;
  if (count <= 0) return [];

  const data = sheet.getRange(startFromBottom, 1, count, sheet.getLastColumn()).getValues();

  return data.reverse().map((row, i) => {
    const o = {_sheetRow: endRow - i};
    headers.forEach((h, j) => { if (h) o[h] = row[j]; });
    return normalizeHistoryRecord_(o);
  }).filter(r => r['Product Name']);
}

function ensureHistoryColumns_(sheet) {
  if (!sheet) return;
  sheet.setRowHeight(1, 36);
  const expectedCompact = [
    'Date','Time','Type','Product Name','Brand','Serial / LOT','Action',
    'Location / Recipient','Recorded By','Operator','Purpose'
  ];
  const width = Math.max(sheet.getLastColumn(), expectedCompact.length);
  const headers = sheet.getRange(HDR_ROW, 1, 1, width).getValues()[0];
  let changed = false;

  // Repair the existing 11-column History layout without adding new columns.
  expectedCompact.forEach((name, i) => {
    if (String(headers[i] || '').trim() !== name) {
      headers[i] = name;
      changed = true;
    }
  });

  if (changed) sheet.getRange(HDR_ROW, 1, 1, width).setValues([headers]);

  // Remove the duplicate block that was briefly created to the right of the old layout.
  const duplicate = ['Location / Recipient','Recorded By','Operator','Note','Reference'];
  const dupHeaders = sheet.getRange(HDR_ROW, 12, 1, duplicate.length).getValues()[0]
    .map(h => String(h || '').trim());
  if (dupHeaders.every((h, i) => h === duplicate[i])) {
    sheet.deleteColumns(12, duplicate.length);
  }

  ensureHistoryExtraColumns_(sheet, historyExtraHeaders_());
  styleHistoryTitle_(sheet, Math.max(sheet.getLastColumn(), expectedCompact.length + historyExtraHeaders_().length));
}

function ensureHistoryExtraColumns_(sheet, names) {
  const width = Math.max(sheet.getLastColumn(), expectedHistoryMinWidth_());
  const headers = sheet.getRange(HDR_ROW, 1, 1, width).getValues()[0]
    .map(h => String(h || '').trim());
  const missing = names.filter(name => !headers.includes(name));
  if (!missing.length) return;

  const startCol = sheet.getLastColumn() + 1;
  sheet.getRange(HDR_ROW, startCol, 1, missing.length).setValues([missing]);
  sheet.getRange(HDR_ROW, startCol, 1, missing.length)
    .setBackground('#4a3580')
    .setFontFamily('Arial').setFontSize(10).setFontWeight('bold').setFontColor('#FFFFFF')
    .setHorizontalAlignment('center').setVerticalAlignment('middle')
    .setBorder(true,true,true,true,true,true,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID);
  missing.forEach((_, i) => sheet.setColumnWidth(startCol + i, 12 * 7));
}

function normalizeHistoryRecord_(o) {

}

function readMonthlyOutSummary(ss, year, month) {
  const sheet = ss.getSheetByName(TABS.history);
  if (!sheet) return {year: year, month: month, totalQty: 0, rows: []};
  ensureHistoryColumns_(sheet);

  const last = sheet.getLastRow();
  if (last < DATA_START) return {year: year, month: month, totalQty: 0, rows: []};

  const width = sheet.getLastColumn();
  const headers = sheet.getRange(HDR_ROW, 1, 1, width).getValues()[0]
    .map(h => String(h || '').trim());
  const data = sheet.getRange(DATA_START, 1, last - DATA_START + 1, width).getValues();
  const selectedKey = monthKey_(year, month);
  const prevKeys = previousMonthKeys_(year, month, 3);
  const lastYearKey = monthKey_(year - 1, month);
  const byProduct = {};

  data.forEach(row => {
    const it = {};
    headers.forEach((h, i) => { if (h) it[h] = row[i]; });
    normalizeHistoryRecord_(it);

    if (String(it['Action'] || '').trim() !== 'Stock Out') return;
    const d = parseHistoryDate_(it['Date']);
    if (!d) return;

    const product = String(it['Product Name'] || '').trim() || '-';
    const brand = String(it['Brand'] || '').trim();
    const type = String(it['Type'] || '').trim();
    const ref = String(it['Reference'] || '').trim();
    const inferred = inferHistoryModelSize_(it);
    const model = String(it['Model'] || inferred.model || '').trim();
    const size = String(it['Size'] || inferred.size || '').trim();
    const isMask = String(type).toLowerCase().indexOf('mask') !== -1;
    const key = [type, product, brand, ref, isMask ? model : '', isMask ? size : ''].join('|');
    const qty = historyOutQty_(it);
    if (!qty) return;

    if (!byProduct[key]) {
      byProduct[key] = {type: type, productName: product, brand: brand, reference: ref, model: model, size: size, months: {}};
    }
    const mk = monthKey_(d.getFullYear(), d.getMonth() + 1);
    byProduct[key].months[mk] = (byProduct[key].months[mk] || 0) + qty;
  });

  const rows = Object.keys(byProduct).map(k => {
    const item = byProduct[k];
    const selectedQty = item.months[selectedKey] || 0;
    const prevTotal = prevKeys.reduce((n, mk) => n + (item.months[mk] || 0), 0);
    const prevAvg = Math.round((prevTotal / prevKeys.length) * 10) / 10;
    const sameMonthLastYearQty = item.months[lastYearKey] || 0;
    const forecastQty = Math.ceil(Math.max(selectedQty, prevAvg, sameMonthLastYearQty));
    return {
      type: item.type,
      productName: item.productName,
      brand: item.brand,
      reference: item.reference,
      model: item.model,
      size: item.size,
      qty: selectedQty,
      prevAvg: prevAvg,
      sameMonthLastYearQty: sameMonthLastYearQty,
      forecastQty: forecastQty
    };
  }).filter(r => r.qty || r.prevAvg || r.sameMonthLastYearQty || r.forecastQty).sort((a, b) =>
    b.forecastQty - a.forecastQty ||
    b.qty - a.qty ||
    String(a.type).localeCompare(String(b.type)) ||
    String(a.productName).localeCompare(String(b.productName))
  );
  return {year: year, month: month, totalQty: rows.reduce((n, r) => n + r.qty, 0), rows: rows};
}

function monthKey_(year, month) {
  return String(year) + '-' + String(month).padStart(2, '0');
}

function previousMonthKeys_(year, month, count) {
  const out = [];
  for (let i = 1; i <= count; i++) {
    const d = new Date(year, month - 1 - i, 1);
    out.push(monthKey_(d.getFullYear(), d.getMonth() + 1));
  }
  return out;
}

function historyDateMatchesMonth_(value, year, month) {
  const d = parseHistoryDate_(value);
  return !!d && d.getFullYear() === year && (d.getMonth() + 1) === month;
}

function parseHistoryDate_(value) {
  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value.getTime())) return value;
  const raw = String(value || '').trim();
  if (!raw) return null;

  const iso = raw.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (iso) return new Date(Number(iso[1]), Number(iso[2]) - 1, Number(iso[3]));

  const slash = raw.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if (!slash) return null;
  let y = Number(slash[3]);
  if (y > 2400) y -= 543;
  return new Date(y, Number(slash[2]) - 1, Number(slash[1]));
}

function historyOutQty_(it) {
  const q = Number(it['Qty']);
  return Number.isFinite(q) && q > 0 ? q : 1;
}

function updateSalesDetails_(ss, b) {
  const sheet = ss.getSheetByName(TABS.history);
  if (!sheet) throw new Error('History sheet not found');
  ensureHistoryColumns_(sheet);

  const row = parseInt(b.historyRow || b.sheetRow || 0, 10);
  if (!row || row < DATA_START || row > sheet.getLastRow()) throw new Error('Invalid history row');

  const headers = sheet.getRange(HDR_ROW, 1, 1, sheet.getLastColumn()).getValues()[0]
    .map(h => String(h || '').trim());
  const col = name => headers.indexOf(name) + 1;
  const rowValues = sheet.getRange(row, 1, 1, sheet.getLastColumn()).getValues()[0];
  const current = {};
  headers.forEach((h, i) => { if (h) current[h] = rowValues[i]; });
  normalizeHistoryRecord_(current);
  const inferred = inferHistoryModelSize_(current);
  const updates = {
    'Buyer Name': b.buyerName || '',
    'Hospital': b.hospital || '',
    'PO No': b.poNo || '',
    'Sale Date': b.saleDate || '',
    'Sale Price': b.salePrice || '',
    'Wholesale Price': b.wholesalePrice || '',
    'Model': b.model || current['Model'] || inferred.model || '',
    'Size': b.size || current['Size'] || inferred.size || ''
  };

  Object.keys(updates).forEach(name => {
    const c = col(name);
    if (c) sheet.getRange(row, c).setValue(updates[name]);
  });
  return {row: row};
}

function inferHistoryModelSize_(it) {
  const currentModel = String(it['Model'] || '').trim();
  const currentSize = String(it['Size'] || '').trim();
  if (currentModel && currentSize) return {model: currentModel, size: currentSize};

  const type = String(it['Type'] || '').toLowerCase();
  const lot = String(it['Serial / LOT'] || it['LOT Number'] || it['LOT'] || '').trim();
  const inferred = inferProductFromLot_(lot) || {};
  const inferredType = String(inferred.type || '').toLowerCase();
  if (type && type.indexOf('mask') === -1 && inferredType !== 'mask') {
    return {model: currentModel, size: currentSize};
  }
  let model = currentModel || inferred.model || '';
  let size = currentSize || inferred.size || '';

  const name = String(it['Product Name'] || '').toUpperCase();
  if (!model) {
    if (name.indexOf('NM6A') !== -1) model = 'EaseFit NM6A';
    else if (name.indexOf('NM6') !== -1) model = 'EaseFit NM6';
    else if (name.indexOf('NM5A') !== -1) model = 'EaseFit NM5A';
    else if (name.indexOf('NM5') !== -1) model = 'EaseFit NM5';
    else if (name.indexOf('PMI') !== -1) model = 'EaseFit PMI';
  }
  if (!size) {
    const sizeMatch = name.match(/\b([SML])\b|NM\dA?([SML])$/);
    size = sizeMatch ? (sizeMatch[1] || sizeMatch[2] || '') : '';
  }
  return {model: model, size: size};
}

function readSalesDetailAlerts(ss) {
  const sheet = ss.getSheetByName(TABS.history);
  if (!sheet) return [];
  ensureHistoryColumns_(sheet);
  const last = sheet.getLastRow();
  if (last < DATA_START) return [];

  const headers = sheet.getRange(HDR_ROW, 1, 1, sheet.getLastColumn()).getValues()[0]
    .map(h => String(h || '').trim());
  const data = sheet.getRange(DATA_START, 1, last - DATA_START + 1, sheet.getLastColumn()).getValues();
  const today = new Date();
  today.setHours(0,0,0,0);
  const alerts = [];

  data.forEach((row, i) => {
    const it = {_sheetRow: i + DATA_START};
    headers.forEach((h, j) => { if (h) it[h] = row[j]; });
    normalizeHistoryRecord_(it);
    if (String(it['Action'] || '').trim() !== 'Stock Out') return;
    const d = parseHistoryDate_(it['Date']);
    if (!d) return;
    d.setHours(0,0,0,0);
    const ageDays = Math.floor((today - d) / 86400000);

    const missingSale = salesDetailMissing_(it);
    const missingPo = !String(it['PO No'] || '').trim();
    if (missingSale && ageDays >= 7) {
      alerts.push(salesAlertRow_(it, ageDays, 'sales_detail', 'Sales detail is missing for more than 7 days'));
    }
    if (missingPo && ageDays >= 30) {
      alerts.push(salesAlertRow_(it, ageDays, 'po', 'PO number is missing for more than 30 days'));
    }
  });

  return alerts.sort((a,b) => b.ageDays - a.ageDays).slice(0, 50);
}

function salesDetailMissing_(it) {
  return !String(it['Buyer Name'] || '').trim() ||
    !String(it['Hospital'] || '').trim() ||
    !String(it['Sale Date'] || '').trim() ||
    !String(it['Sale Price'] || '').trim() ||
    !String(it['Wholesale Price'] || '').trim();
}

function salesAlertRow_(it, ageDays, kind, message) {
  return {
    historyRow: it._sheetRow,
    kind: kind,
    message: message,
    ageDays: ageDays,
    date: it['Date'] || '',
    productName: it['Product Name'] || '',
    brand: it['Brand'] || '',
    model: it['Model'] || '',
    size: it['Size'] || '',
    serialLot: it['Serial / LOT'] || '',
    recipient: it['Location / Recipient'] || '',
    buyerName: it['Buyer Name'] || '',
    hospital: it['Hospital'] || '',
    poNo: it['PO No'] || '',
    saleDate: it['Sale Date'] || '',
    salePrice: it['Sale Price'] || '',
    wholesalePrice: it['Wholesale Price'] || ''
  };
}

function normalizeText(v) {
  return String(v || '').trim();
}

function getUserRole(user) {
  return ADMIN_USERS[user] ? 'admin' : 'stock';
}

function getAuthSecret() {
  const props = PropertiesService.getScriptProperties();
  let secret = props.getProperty('SKL_AUTH_SECRET');
  if (!secret) {
    secret = Utilities.getUuid() + ':' + Utilities.getUuid();
    props.setProperty('SKL_AUTH_SECRET', secret);
  }
  return secret;
}

function makeAuthToken(user, staffName) {
  const secret = getAuthSecret();
  const subject = String(user) + '|' + String(staffName || '').trim();
  const sig = Utilities.computeHmacSha256Signature(subject, secret);
  return Utilities.base64EncodeWebSafe(sig);
}

function requireAuth(payload, requiredRole) {
  const user = String(payload.user || '');
  const staffName = String(payload.staffName || payload.operator || '').trim();
  const token = String(payload.token || '');
  if (!USERS[user] || !staffName || !token || token !== makeAuthToken(user, staffName)) {
    return { ok:false, error:'Unauthorized or missing staff name' };
  }
  if (requiredRole === 'admin' && getUserRole(user) !== 'admin') {
    return { ok:false, error:'Admin permission required' };
  }
  return { ok:true, user:user, role:getUserRole(user), staffName:staffName };
}

function requiresWriteLock_(action) {
  return [
    'add_product',
    'in',
    'out',
    'out_by_id',
    'batch_stocktake',
    'update_product_names',
    'update_sales_details',
    'sort'
  ].indexOf(String(action || '')) !== -1;
}

function normalizeLot(lot) {
  lot = String(lot || '').trim().toUpperCase();
  if (!lot) return '';

  const idxBy = lot.indexOf('BY-');
  if (idxBy !== -1) return lot.substring(idxBy);


  return lot.replace(/^\d{10,}/, '').replace(/C\.\d+(?:\.\d+)+/g, '').trim();
}

function findExistingAggregateRow(sheet, type, b) {
  // Masks are stocked by product + size + LOT so expiry and batch-level stock
  // stay accurate. Accessories keep the older product-level aggregation.
  const last = sheet.getLastRow();
  if (last < DATA_START) return -1;

  const numCols = sheet.getLastColumn();
  const data = sheet.getRange(DATA_START, 1, last - DATA_START + 1, numCols).getValues();

  const sizeCol = MASK_COL_SIZE;

  for (let i = 0; i < data.length; i++) {
    const row = data[i];
    if (normalizeText(row[COL_STATUS - 1]) !== 'Active') continue;

    const productName = normalizeText(row[3]);   // col 4
    const brand       = normalizeText(row[5]);   // col 6
    const size        = type === 'mask' ? normalizeText(row[sizeCol - 1]) : '';
    const lot         = type === 'mask' ? normalizeLot(row[MASK_COL_LOT - 1]) : '';

    const sameName  = productName === normalizeText(b.name);
    const sameBrand = brand       === normalizeText(b.brand);
    const sameSize  = (type !== 'mask') || (size === normalizeText(b.size));
    const sameLot   = (type !== 'mask') || (lot === normalizeLot(b.lot));

    if (sameName && sameBrand && sameSize && sameLot) {
      return i + DATA_START;
    }
  }

  return -1;
}

function rowMatchesPayload(row, type, b) {
  if (normalizeText(row[COL_STATUS - 1]).toLowerCase() !== 'active') return false;

  // When a trusted sheet row is supplied, the row itself is the identity.
  // The remaining payload fields are user-facing metadata and may differ in
  // formatting from the sheet, especially for manual stock-out.
  if (b && b.sheetRow) return true;

  const rowName = normalizeText(row[3]).toUpperCase();
  const rowBrand = normalizeText(row[5]).toUpperCase();
  if (b.name && rowName !== normalizeText(b.name).toUpperCase()) return false;
  if (b.brand && rowBrand !== normalizeText(b.brand).toUpperCase()) return false;

  if (type === 'device') {
    const rowSn = normalizeText(row[DEV_COL_SN - 1]).toUpperCase();
    const rowLot = normalizeLot(row[DEV_COL_LOT - 1]);
    const rowGtin = normalizeText(row[9]).replace(/\s/g, '').replace(/^0+/, '').toUpperCase(); // col 10
    const sn = normalizeText(b.sn).toUpperCase();
    const lot = normalizeLot(b.lot);
    const gtin = normalizeText(b.gtin).replace(/\s/g, '').replace(/^0+/, '').toUpperCase();
    if (sn) return rowSn === sn;
    if (lot) return rowLot === lot;
    if (gtin) return rowGtin === gtin;
    return false;
  }

  const lotCol = type === 'mask' ? MASK_COL_LOT : ACC_COL_LOT;
  const gtinCol = type === 'mask' ? MASK_COL_GTIN : ACC_COL_GTIN;
  const rowLot = normalizeLot(row[lotCol - 1]);
  const rowGtin = normalizeText(row[gtinCol - 1]).replace(/\s/g, '').replace(/^0+/, '').toUpperCase();
  const lot = normalizeLot(b.lot);
  const gtin = normalizeText(b.gtin).replace(/\s/g, '').replace(/^0+/, '').toUpperCase();
  if (lot && rowLot !== lot) return false;
  if (!lot && gtin && rowGtin !== gtin) return false;
  if (!lot && !gtin) return false;

  if (type === 'mask' && b.size) {
    const rowSize = normalizeText(row[MASK_COL_SIZE - 1]).toUpperCase();
    if (rowSize !== normalizeText(b.size).toUpperCase()) return false;
  }
  return true;
}

function styleDataRow(sheet, r, type) {
  const last = sheet.getLastColumn();
  sheet.getRange(r,1,1,last)
    .setFontFamily('Arial')
    .setFontSize(10)
    .setFontWeight('normal')
    .setVerticalAlignment('middle')
    .setBackground(r % 2 === 0 ? '#f7f8fa' : '#FFFFFF')
    .setBorder(true,true,true,true,true,true,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID);

  const clrs = {
    device:['#dce8f7','#1a4fa0'],
    mask:['#d6f0e3','#1a7a4a'],
    accessory:['#fdf3d0','#a06a00']
  };

  const c = clrs[type] || clrs.device;
  sheet.getRange(r, COL_STATUS)
    .setBackground(c[0])
    .setFontColor(c[1])
    .setFontWeight('bold');
}

function markOut(sheet, r, type, date) {
  sheet.getRange(r, COL_STATUS).setValue('Out');

  if (type === 'device') {
    sheet.getRange(r, DEV_COL_OUT_DATE).setValue(date);
  }

  sheet.getRange(r,1,1,sheet.getLastColumn()).setBackground('#fde8e8');
  sheet.getRange(r, COL_STATUS).setFontColor('#b92e2e').setFontWeight('bold');
}

function formatOutDetail_(location, purpose, note) {
  const parts = [];
  if (location) parts.push('OUT TO: ' + location);
  if (purpose) parts.push('PURPOSE: ' + purpose);
  if (note) parts.push(note);
  return parts.join(' | ');
}

function appendHistory(ss, b, date, time, action) {
  const sheet = ss.getSheetByName(TABS.history);
  if (!sheet) return;
  ensureHistoryColumns_(sheet);

  const tl = {device:'Device', mask:'Mask', accessory:'Accessory'};
  const snLot = b.sn ? 'SN: ' + b.sn : (b.lot ? 'LOT: ' + b.lot : (b.productCode ? 'CODE: ' + b.productCode : ''));
  const qty = Math.max(1, parseInt(b.qty || 1, 10));

  const valueByHeader = {
    'Date': date,
    'Time': time,
    'Type': tl[b.type] || b.type,
    'Product Name': b.name || '',
    'Brand': b.brand || '',
    'Model': b.model || '',
    'Size': b.size || '',
    'Serial / LOT': snLot,
    'Action': action,
    'Qty': qty,
    'Before Qty': Number.isFinite(Number(b.beforeQty)) ? Number(b.beforeQty) : '',
    'After Qty': Number.isFinite(Number(b.afterQty)) ? Number(b.afterQty) : '',
    'Purpose': b.purpose || '',
    'Location / Recipient': b.recipient || b.location || '',
    'Recipient': b.recipient || b.location || '',
    'Recorded By': b.user || '',
    'Operator': b.operator || b.staffName || b.user || '',
    'Note': b.note || '',
    'Reference': b.ref || '',
    'Buyer Name': b.buyerName || '',
    'Hospital': b.hospital || '',
    'PO No': b.poNo || '',
    'Sale Date': b.saleDate || '',
    'Sale Price': b.salePrice || '',
    'Wholesale Price': b.wholesalePrice || ''
  };
  const headers = sheet.getRange(HDR_ROW, 1, 1, sheet.getLastColumn()).getValues()[0]
    .map(h => String(h || '').trim());
  sheet.appendRow(headers.map(h => valueByHeader[h] == null ? '' : valueByHeader[h]));

  const r = sheet.getLastRow();
  sheet.getRange(r,1,1,sheet.getLastColumn())
    .setBackground(r % 2 === 0 ? '#f7f8fa' : '#FFFFFF')
    .setFontFamily('Arial')
    .setFontSize(10)
    .setBorder(true,true,true,true,true,true,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID);

  const ac = sheet.getRange(r,7);
  if (action === 'Stock In') {
    ac.setBackground('#d6f0e3').setFontColor('#1a7a4a').setFontWeight('bold');
  } else if (action.indexOf('Stocktake') !== -1) {
    ac.setBackground('#e8eefc').setFontColor('#2753a7').setFontWeight('bold');
  } else {
    ac.setBackground('#fde8e8').setFontColor('#b92e2e').setFontWeight('bold');
  }
}

function validateTransaction_(b, action) {
  if (!b) throw new Error('Missing payload');
  if (!b.type || ['device','mask','accessory'].indexOf(String(b.type)) === -1) throw new Error('Invalid product type');
  if (!String(b.name || '').trim()) throw new Error('Missing product name');
  const qty = parseInt(b.qty || 1, 10);
  if (!qty || qty <= 0) throw new Error('Quantity must be greater than 0');
  if (!String(b.operator || b.staffName || b.user || '').trim()) throw new Error('Missing operator');
  if (b.type === 'device' && !String(b.sn || '').trim() && !(action === 'out' && b.sheetRow)) throw new Error('Missing serial number (SN)');
  if (action !== 'out' && (b.type === 'mask' || b.type === 'accessory') && !String(b.lot || '').trim()) throw new Error('Missing lot number (LOT)');
  if (action === 'out') {
    const purpose = String(b.purpose || '').trim();
    const recipient = String(b.recipient || b.location || '').trim();
    if (!purpose) throw new Error('Missing stock-out purpose');
    if (purpose === '\u0e2d\u0e37\u0e48\u0e19\u0e46' && !String(b.purposeOther || '').trim()) throw new Error('Please specify other purpose');
    if (!recipient) throw new Error('Missing recipient');
    b.purpose = purpose === '\u0e2d\u0e37\u0e48\u0e19\u0e46' ? '\u0e2d\u0e37\u0e48\u0e19\u0e46: ' + String(b.purposeOther || '').trim() : purpose;
    b.recipient = recipient;
  }
}

function applyAuthContext_(body, auth) {
  body.user = auth.user;
  body.operator = String(body.staffName || body.operator || auth.staffName || auth.user || '').trim();
  return body;
}

 * ========================================================= */
function doStockIn(ss, b, date, time) {
  const sheet = ss.getSheetByName(TABS[b.type]);
  if (!sheet) throw new Error('Sheet not found: ' + b.type);

  b.lot = normalizeLot(b.lot);


  if (b.type === 'device') {
    const seq = sheet.getLastRow() - HDR_ROW + 1;
    const row = [
      seq,
      date,
      time,
      b.name || '',
      'Active',
      b.brand || '',
      b.model || '',
      b.sn || '',
      b.lot || '',
      b.gtin || '',
      b.mfgDate || '',
      b.expiry || '',
      b.location || '',
      '',
      b.user || '',
      b.note || ''
    ];

    sheet.appendRow(row);
    const newRow = sheet.getLastRow();
    styleDataRow(sheet, newRow, b.type);
    appendHistory(ss, b, date, time, 'Stock In');
    return newRow;
  }


  const addQty = Math.max(1, parseInt(b.qty || 1, 10));
  const existingRow = findExistingAggregateRow(sheet, b.type, b);

  if (existingRow !== -1) {
    const qtyCol = (b.type === 'mask') ? MASK_COL_QTY : ACC_COL_QTY;
    const oldQty = parseInt(sheet.getRange(existingRow, qtyCol).getValue() || 0, 10);
    const newQty = oldQty + addQty;
    sheet.getRange(existingRow, qtyCol).setValue(newQty);

    if (b.note) {
      const noteCol = (b.type === 'mask') ? MASK_COL_NOTE : ACC_COL_NOTE;
      const oldNote = normalizeText(sheet.getRange(existingRow, noteCol).getValue());
      sheet.getRange(existingRow, noteCol).setValue(oldNote ? oldNote + ' | ' + b.note : b.note);
    }

    if (normalizeText(sheet.getRange(existingRow, COL_STATUS).getValue()) !== 'Active') {
      sheet.getRange(existingRow, COL_STATUS).setValue('Active');
    }

    sheet.getRange(existingRow,1,1,sheet.getLastColumn()).setBackground('#FFFFFF');
    styleDataRow(sheet, existingRow, b.type);
    if (_needsSort(sheet, COL_STATUS, 2)) sortMaskAccSheet(sheet);
    b.beforeQty = oldQty;
    b.afterQty = newQty;
    appendHistory(ss, b, date, time, 'Stock In');
    return {row: existingRow, merged: true, newQty: newQty};
  }

  const seq = sheet.getLastRow() - HDR_ROW + 1;
  let row = [];

  if (b.type === 'mask') {
    row = [
      seq,
      date,
      time,
      b.name || '',
      'Active',
      b.brand || '',
      b.model || '',
      b.size || '',
      b.ref || '',
      b.lot || '',
      b.gtin || '',
      b.mfgDate || '',
      addQty,
      b.note || '',
      b.expiry || ''
    ];
  } else if (b.type === 'accessory') {
    row = [
      seq,
      date,
      time,
      b.name || '',
      'Active',
      b.brand || '',
      b.lot || '',
      b.gtin || '',
      b.mfgDate || '',
      addQty,
      b.location || '',
      b.note || ''
    ];
  }

  sheet.appendRow(row);
  const newRow = sheet.getLastRow();
  styleDataRow(sheet, newRow, b.type);
  if (_needsSort(sheet, COL_STATUS, 2)) sortMaskAccSheet(sheet);
  appendHistory(ss, b, date, time, 'Stock In');
  return newRow;
}

function parseExpiryDate_(v) {
  if (!v) return null;
  const d = (v instanceof Date) ? v : new Date(v);
  if (isNaN(d.getTime())) return null;
  d.setHours(0, 0, 0, 0);
  return d;
}

/* =========================================================
 * STOCK OUT
 * ========================================================= */
function doStockOut(ss, b, date, time) {
  const sheet = ss.getSheetByName(TABS[b.type]);
  if (!sheet) throw new Error('Sheet not found: ' + b.type);

  b.lot = normalizeLot(b.lot);

  const last = sheet.getLastRow();
  if (last < DATA_START) return -1;

  const numCols = sheet.getLastColumn();
  const data = sheet.getRange(DATA_START, 1, last - DATA_START + 1, numCols).getValues();


  if (b.type === 'device') {
    const keyVal = b.sn || b.lot || '';
    const gtin = normalizeText(b.gtin).replace(/\s/g, '').replace(/^0+/, '').toUpperCase();
    for (let i = data.length - 1; i >= 0; i--) {
      const row = data[i];
      if (normalizeText(row[COL_STATUS - 1]) !== 'Active') continue;

      const rowSn  = normalizeText(row[DEV_COL_SN - 1]);
      const rowLot = normalizeText(row[DEV_COL_LOT - 1]);
      const rowGtin = normalizeText(row[9]).replace(/\s/g, '').replace(/^0+/, '').toUpperCase();
      const r = i + DATA_START;

      const snMatch = b.sn && rowSn.toUpperCase() === normalizeText(b.sn).toUpperCase();
      const lotMatch = !b.sn && b.lot && normalizeLot(rowLot) === normalizeLot(keyVal);
      const gtinMatch = !b.sn && !b.lot && gtin && rowGtin === gtin;
      if (snMatch || lotMatch || gtinMatch) {
        markOut(sheet, r, b.type, date);

        if (b.location || b.purpose || b.note) {
          const oldNote = normalizeText(row[15]);  // col 16 = Note
          const extra = formatOutDetail_(b.location || b.recipient || '', b.purpose || '', b.note || '');
          sheet.getRange(r, 16).setValue(oldNote ? oldNote + ' | ' + extra : extra);
        }

        appendHistory(ss, b, date, time, 'Stock Out');
        return r;
      }
    }
    return -1;
  }


  const lotCol  = b.type === 'mask' ? MASK_COL_LOT  : ACC_COL_LOT;
  const qtyCol  = b.type === 'mask' ? MASK_COL_QTY  : ACC_COL_QTY;
  const outQty  = Math.max(1, parseInt(b.qty || 1, 10));

  const matches = [];
  for (let i = 0; i < data.length; i++) {
    const row = data[i];
    if (normalizeText(row[COL_STATUS - 1]) !== 'Active') continue;

    const rowLot  = normalizeText(row[lotCol - 1]);
    const rowName = normalizeText(row[3]);  // col 4 = Product Name
    const rowSize = b.type === 'mask' ? normalizeText(row[MASK_COL_SIZE - 1]) : '';

    if (b.name && rowName.toUpperCase() !== normalizeText(b.name).toUpperCase()) continue;
    if (b.type === 'mask' && b.size && rowSize.toUpperCase() !== normalizeText(b.size).toUpperCase()) continue;
    if (b.type === 'mask' && b.lot && normalizeLot(rowLot) !== normalizeLot(b.lot)) continue;
    if (b.type !== 'mask' && b.lot && normalizeLot(rowLot) !== normalizeLot(b.lot)) continue;

    const qty = parseInt(row[qtyCol - 1] || 0, 10);
    if (qty <= 0) continue;
    matches.push({ row, index: i, sheetRow: i + DATA_START, qty });
  }

  const picked = pickStockOutBatch_(matches, b.type);
  if (picked) {
    const actualOutQty = Math.min(outQty, picked.qty);

    if (picked.qty > actualOutQty) {
      sheet.getRange(picked.sheetRow, qtyCol).setValue(picked.qty - actualOutQty);
    } else {
      markOut(sheet, picked.sheetRow, b.type, date);
    }

    appendOutRow(sheet, b.type, picked.row, date, time, b.location || b.recipient || '', b.note || '', actualOutQty, b.purpose || '');
    b.qty = actualOutQty;
    b.beforeQty = picked.qty;
    b.afterQty = Math.max(0, picked.qty - actualOutQty);
    if (picked.qty <= actualOutQty && _needsSort(sheet, COL_STATUS, 2)) sortMaskAccSheet(sheet);

    appendHistory(ss, b, date, time, 'Stock Out');
    return picked.sheetRow;
  }

  return -1;
}

function pickStockOutBatch_(matches, type) {
  if (!matches.length) return null;
  if (type !== 'mask') return matches[0];
  return matches.slice().sort((a, b) => {
    const da = parseSheetDate_(a.row[1]);
    const db = parseSheetDate_(b.row[1]);
    if (da && db && da.getTime() !== db.getTime()) return da - db;
    if (da) return -1;
    if (db) return 1;
    return a.sheetRow - b.sheetRow;
  })[0];
}

function parseSheetDate_(v) {
  if (!v) return null;
  if (v instanceof Date && !isNaN(v.getTime())) {
    const d = new Date(v.getTime());
    d.setHours(0, 0, 0, 0);
    return d;
  }
  const s = String(v || '').trim();
  let m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if (m) {
    let y = parseInt(m[3], 10);
    if (y > 2400) y -= 543;
    const d = new Date(y, parseInt(m[2], 10) - 1, parseInt(m[1], 10));
    return isNaN(d.getTime()) ? null : d;
  }
  m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
  if (m) {
    const d = new Date(parseInt(m[1], 10), parseInt(m[2], 10) - 1, parseInt(m[3], 10));
    return isNaN(d.getTime()) ? null : d;
  }
  const d = new Date(s);
  if (isNaN(d.getTime())) return null;
  d.setHours(0, 0, 0, 0);
  return d;
}


function appendOutRow(sheet, type, srcRow, date, time, location, note, qty, purpose) {
  qty = Math.max(1, parseInt(qty || 1, 10));
  const seq = sheet.getLastRow() - HDR_ROW + 1;
  let outRow = [];

  if (type === 'mask') {
    outRow = [
      seq,
      date,
      time,
      srcRow[3],              // Product Name
      'Out',
      srcRow[5],              // Brand
      srcRow[6],              // Model
      srcRow[MASK_COL_SIZE - 1],   // Size
      srcRow[MASK_COL_REF  - 1],   // REF
      srcRow[MASK_COL_LOT  - 1],   // LOT
      srcRow[MASK_COL_GTIN - 1],   // GTIN
      srcRow[MASK_COL_MFG    - 1],   // Mfg Date
      qty,
      formatOutDetail_(location, purpose, note),
      srcRow[MASK_COL_EXPIRY - 1] || ''  // Expiry
    ];
  } else if (type === 'accessory') {
    outRow = [
      seq,
      date,
      time,
      srcRow[3],              // Product Name
      'Out',
      srcRow[5],              // Brand
      srcRow[ACC_COL_LOT  - 1],    // LOT
      srcRow[ACC_COL_GTIN - 1],    // GTIN
      srcRow[ACC_COL_MFG  - 1],    // Mfg Date
      qty,
      location,
      formatOutDetail_('', purpose, note)
    ];
  }

  if (!outRow.length) return;

  sheet.appendRow(outRow);
  const newRow = sheet.getLastRow();


  sheet.getRange(newRow, 1, 1, sheet.getLastColumn())
    .setBackground('#fde8e8')
    .setFontFamily('Arial')
    .setFontSize(10)
    .setVerticalAlignment('middle')
    .setBorder(true,true,true,true,true,true,'#d0d0d0',SpreadsheetApp.BorderStyle.SOLID);

  sheet.getRange(newRow, COL_STATUS)
    .setFontColor('#b92e2e')
    .setFontWeight('bold');
}

  range.sort([

    { column: DEV_COL_SN,  ascending: true }
  ]);
}

  const last = sheet.getLastRow();
  if (last < DATA_START + 1) return;
  const range = sheet.getRange(DATA_START, 1, last - DATA_START + 1, sheet.getLastColumn());
  range.sort([

function _needsSort(sheet, statusCol, dateCol) {
  const last = sheet.getLastRow();
  if (last < DATA_START + 1) return false;
  const statuses = sheet.getRange(DATA_START, statusCol, last - DATA_START + 1, 1).getValues();
  let seenOut = false;
  for (let i = 0; i < statuses.length; i++) {
    const status = normalizeText(statuses[i][0]);
    if (status === 'Out') seenOut = true;
    if (status === 'Active' && seenOut) return true;
  }
  return false;
}

function doStockOutByRow(ss, b, date, time) {
  const sheet = ss.getSheetByName(TABS[b.type]);
  if (!sheet) throw new Error('Sheet not found: ' + b.type);

  const r = parseInt(b.sheetRow, 10);
  if (!r || r < DATA_START || r > sheet.getLastRow()) {
    throw new Error('Invalid row');
  }
  const numCols = sheet.getLastColumn();
  const srcRow = sheet.getRange(r, 1, 1, numCols).getValues()[0];

  if (!rowMatchesPayload(srcRow, b.type, b)) {
    throw new Error('Row does not match scanned item');
  }

  const outQty = Math.max(1, parseInt(b.qty || 1, 10));

  if (b.type === 'device') {
    markOut(sheet, r, b.type, date);
    if (b.location || b.purpose || b.note) {
      const oldNote = normalizeText(srcRow[15]);
      const extra = formatOutDetail_(b.location || b.recipient || '', b.purpose || '', b.note || '');
      sheet.getRange(r, 16).setValue(oldNote ? oldNote + ' | ' + extra : extra);
    }

  } else {

    const qtyCol = b.type === 'mask' ? MASK_COL_QTY : ACC_COL_QTY;
    const qty = parseInt(srcRow[qtyCol - 1] || 0, 10);
    if (qty <= 0) throw new Error('No stock left');
    const actualOutQty = Math.min(outQty, qty);

    if (qty > actualOutQty) {
      sheet.getRange(r, qtyCol).setValue(qty - actualOutQty);
    } else {
      markOut(sheet, r, b.type, date);
    }
    appendOutRow(sheet, b.type, srcRow, date, time, b.recipient || b.location || '', b.note || '', actualOutQty, b.purpose || '');
    b.qty = actualOutQty;
    b.beforeQty = qty;
    b.afterQty = Math.max(0, qty - actualOutQty);
    if (qty <= actualOutQty && _needsSort(sheet, COL_STATUS, 2)) sortMaskAccSheet(sheet);
  }

  appendHistory(ss, b, date, time, 'Stock Out');
}

function findFifoMaskRowForSelected_(sheet, selectedRow) {
  const last = sheet.getLastRow();
  if (last < DATA_START) return null;
  const data = sheet.getRange(DATA_START, 1, last - DATA_START + 1, sheet.getLastColumn()).getValues();
  const matches = [];
  data.forEach((row, i) => {
    if (normalizeText(row[COL_STATUS - 1]) !== 'Active') return;
    if (parseInt(row[MASK_COL_QTY - 1] || 0, 10) <= 0) return;
    if (!sameMaskStockIdentity_(row, selectedRow)) return;
    matches.push({
      row: row,
      index: i,
      sheetRow: i + DATA_START,
      qty: parseInt(row[MASK_COL_QTY - 1] || 0, 10)
    });
  });
  return pickStockOutBatch_(matches, 'mask');
}

function sameMaskStockIdentity_(a, b) {
  return normalizeText(a[3]).toUpperCase() === normalizeText(b[3]).toUpperCase() &&
    normalizeText(a[5]).toUpperCase() === normalizeText(b[5]).toUpperCase() &&
    normalizeText(a[6]).toUpperCase() === normalizeText(b[6]).toUpperCase() &&
    normalizeText(a[MASK_COL_SIZE - 1]).toUpperCase() === normalizeText(b[MASK_COL_SIZE - 1]).toUpperCase() &&
    normalizeText(a[MASK_COL_REF - 1]).toUpperCase() === normalizeText(b[MASK_COL_REF - 1]).toUpperCase();
}

function doBatchStocktake(ss, items, date, time, ctx) {
  if (!Array.isArray(items) || !items.length) throw new Error('No stocktake items');
  if (!String(ctx.operator || ctx.staffName || '').trim()) throw new Error('Missing operator');
  const products = readProductDB(ss);
  const byCode = buildUniqueProductCodeMap_(products);
  let updated = 0;

  items.forEach(item => {
    const qty = parseInt(item.qty, 10);
    if (!Number.isFinite(qty) || qty < 0) throw new Error('Invalid stocktake quantity');

    let product = null;
    let result = null;
    const exact = stocktakeBySheetRow_(ss, item, qty);
    if (exact) {
      product = exact.product;
      result = exact.result;
    } else {
      const code = String(item.productCode || item.product_code || '').trim().toUpperCase();
      if (!code) throw new Error('Invalid stocktake item');
      product = resolveStocktakeProduct_(products, byCode, item, code);
      if (!product) throw new Error('Product code not found or ambiguous: ' + code);
      if (product.type === 'device') return;
      const sheet = ss.getSheetByName(TABS[product.type]);
      if (!sheet) throw new Error('Sheet not found: ' + product.type);
      result = setAggregateQty_(sheet, product, qty, date, ctx);
    }
    if (!product || product.type === 'device') return;
    appendHistory(ss, {
      type: product.type, name: product.name, brand: product.brand, model: product.model, size: product.size,
      ref: product.ref, productCode: product.product_code, lot: product.lot, gtin: product.gtin,
      qty: qty, beforeQty: result.beforeQty, afterQty: result.afterQty,
      purpose: 'ปรับยอดจาก stocktake', recipient: '', user: ctx.user, operator: ctx.operator, note: 'batch stocktake'
    }, date, time, 'Stocktake');
    updated++;
  });
  return {updated:updated};
}

function stocktakeBySheetRow_(ss, item, qty) {
  const type = normalizeType_(item.type, '', item.name);
  if (type !== 'mask' && type !== 'accessory') return null;
  const sheetRow = parseInt(item.sheetRow, 10);
  if (!Number.isFinite(sheetRow) || sheetRow < DATA_START) return null;
  const sheet = ss.getSheetByName(TABS[type]);
  if (!sheet || sheetRow > sheet.getLastRow()) throw new Error('Stocktake row not found');
  const qtyCol = type === 'mask' ? MASK_COL_QTY : ACC_COL_QTY;
  const row = sheet.getRange(sheetRow, 1, 1, sheet.getLastColumn()).getValues()[0];
  const beforeQty = parseInt(row[qtyCol - 1] || 0, 10);
  sheet.getRange(sheetRow, qtyCol).setValue(qty);
  sheet.getRange(sheetRow, COL_STATUS).setValue(qty > 0 ? 'Active' : 'Out');
  styleDataRow(sheet, sheetRow, type);
  return {
    product: productFromStockRow_(type, row),
    result: {row:sheetRow, beforeQty:beforeQty, afterQty:qty}
  };
}

function productFromStockRow_(type, row) {
  if (type === 'mask') {
    return {
      type:'mask',
      name: normalizeText(row[3]),
      brand: normalizeText(row[5]),
      model: normalizeText(row[6]),
      size: normalizeText(row[MASK_COL_SIZE - 1]),
      ref: normalizeText(row[MASK_COL_REF - 1]),
      lot: normalizeText(row[MASK_COL_LOT - 1]),
      gtin: normalizeText(row[MASK_COL_GTIN - 1]),
      product_code: normalizeText(row[MASK_COL_REF - 1] || row[MASK_COL_GTIN - 1])
    };
  }
  return {
    type:'accessory',
    name: normalizeText(row[3]),
    brand: normalizeText(row[5]),
    model:'',
    size:'',
    ref:'',
    lot: normalizeText(row[ACC_COL_LOT - 1]),
    gtin: normalizeText(row[ACC_COL_GTIN - 1]),
    product_code: normalizeText(row[ACC_COL_GTIN - 1])
  };
}

function resolveStocktakeProduct_(products, byCode, item, code) {
  const direct = byCode.get(code);
  if (direct) return direct;
  const matches = (products || []).filter(p => String(p.product_code || '').trim().toUpperCase() === code);
  if (!matches.length) return null;
  const hinted = matches.filter(p =>
    (!item.type || normalizeType_(p.type, p.category, p.name) === normalizeType_(item.type, '', item.name)) &&
    (!item.name || sameProductIdentityText_(p.name || p.product_name, item.name)) &&
    (!item.brand || sameProductIdentityText_(p.brand, item.brand)) &&
    (!item.model || sameProductIdentityText_(p.model, item.model)) &&
    (!item.size || sameProductIdentityText_(p.size, item.size)) &&
    (!item.ref || sameProductIdentityText_(p.ref, item.ref))
  );
  return hinted.length === 1 ? hinted[0] : null;
}

function buildUniqueProductCodeMap_(products) {
  const buckets = {};
  (products || []).forEach(p => {
    const code = String(p.product_code || '').trim().toUpperCase();
    if (!code) return;
    if (!buckets[code]) buckets[code] = [];
    buckets[code].push(p);
  });
  const out = new Map();
  Object.keys(buckets).forEach(code => {
    if (buckets[code].length === 1) {
      out.set(code, buckets[code][0]);
    } else {
      Logger.log('Duplicate Product Code skipped for stocktake: ' + code + ' (' + buckets[code].length + ' rows)');
    }
  });
  return out;
}

function setAggregateQty_(sheet, product, qty, date, ctx) {
  const type = product.type;
  const rowPayload = {
    type:type, name:product.name, brand:product.brand, model:product.model, size:product.size, ref:product.ref,
    lot:product.lot || '', gtin:product.gtin || '', qty:qty, location:'Stocktake', note:'stocktake by ' + (ctx.operator || ctx.user || '')
  };
  const row = findExistingAggregateRow(sheet, type, rowPayload);
  const qtyCol = type === 'mask' ? MASK_COL_QTY : ACC_COL_QTY;
  if (row !== -1) {
    const beforeQty = parseInt(sheet.getRange(row, qtyCol).getValue() || 0, 10);
    sheet.getRange(row, qtyCol).setValue(qty);
    sheet.getRange(row, COL_STATUS).setValue(qty > 0 ? 'Active' : 'Out');
    styleDataRow(sheet, row, type);
    return {row:row, beforeQty:beforeQty, afterQty:qty};
  }
  if (qty <= 0) return {row:-1, beforeQty:0, afterQty:0};
  const seq = sheet.getLastRow() - HDR_ROW + 1;
  const newRow = type === 'mask'
    ? [seq,date,'',product.name,'Active',product.brand,product.model,product.size,product.ref,product.lot || '',product.gtin || '','',qty,'stocktake create','']
    : [seq,date,'',product.name,'Active',product.brand,'',product.gtin || '','',qty,'Stocktake','stocktake create'];
  sheet.appendRow(newRow);
  const r = sheet.getLastRow();
  styleDataRow(sheet, r, type);
  return {row:r, beforeQty:0, afterQty:qty};
}

function checkAndAlert() {
  const email = ALERT_EMAIL.trim();
  if (!email) { Logger.log('Missing ALERT_EMAIL'); return; }

  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const today = new Date(); today.setHours(0,0,0,0);
  const { lowItems, expItems } = _collectAlertItems(ss, today);

  if (!lowItems.length && !expItems.length) {
    Logger.log('No alert items found');
    return;
  }


  const props      = PropertiesService.getScriptProperties();
  const currentKey = JSON.stringify({
    low: lowItems.map(x => x.name + '|' + x.size),
    exp: expItems.map(x => x.sn  + '|' + x.expDate),
  });
  if (currentKey === (props.getProperty('lastAlertKey') || '')) {
    Logger.log('Alerts already sent today; skipping duplicate send');
    return;
  }
  props.setProperty('lastAlertKey', currentKey);

  const subject = `Skyline Stock Alert - ${lowItems.length} low stock, ${expItems.length} near expiry`;
  MailApp.sendEmail({ to: email, subject, htmlBody: _buildAlertHtml(lowItems, expItems, today) });
  Logger.log('Stock alert sent to ' + email);
}


function _collectAlertItems(ss, today) {
  const lowItems = [];
  const expItems = [];
  const warnMs   = ALERT_EXP_DAYS * 86400000;


  const maskSheet = ss.getSheetByName(TABS.mask);
  if (maskSheet && maskSheet.getLastRow() >= DATA_START) {
    const data = maskSheet.getRange(
      DATA_START, 1,
      maskSheet.getLastRow() - DATA_START + 1,
      maskSheet.getLastColumn()
    ).getValues();
    data.forEach(row => {
      if (String(row[COL_STATUS - 1] || '').trim() !== 'Active') return;
      const qty = Number(row[MASK_COL_QTY - 1] || 0);
      if (qty <= ALERT_LOW_QTY) {
        lowItems.push({
          type: 'Mask',
          name: String(row[3] || ''),
          size: String(row[MASK_COL_SIZE - 1] || ''),
          qty,
        });
      }
    });
  }



  if (maskSheet && maskSheet.getLastRow() >= DATA_START) {
    const numColsMask = maskSheet.getLastColumn();
    if (numColsMask >= MASK_COL_EXPIRY) {
      const maskData = maskSheet.getRange(
        DATA_START, 1,
        maskSheet.getLastRow() - DATA_START + 1,
        numColsMask
      ).getValues();
      maskData.forEach(row => {
        if (String(row[COL_STATUS - 1] || '').trim() !== 'Active') return;
        const expRaw = row[MASK_COL_EXPIRY - 1];
        if (!expRaw) return;
        const expDate = (expRaw instanceof Date) ? expRaw : new Date(expRaw);
        if (isNaN(expDate.getTime())) return;
        expDate.setHours(0,0,0,0);
        const daysLeft = Math.ceil((expDate - today) / 86400000);
        if (daysLeft <= ALERT_EXP_DAYS) {
          expItems.push({
            type:    'Mask',
            name:    String(row[3] || ''),
            sn:      String(row[MASK_COL_LOT - 1] || ''),
            daysLeft,
            expDate: Utilities.formatDate(expDate, TZ, 'dd/MM/yyyy'),
          });
        }
      });
    }
  }


  const accSheet = ss.getSheetByName(TABS.accessory);
  if (accSheet && accSheet.getLastRow() >= DATA_START) {
    const data = accSheet.getRange(
      DATA_START, 1,
      accSheet.getLastRow() - DATA_START + 1,
      accSheet.getLastColumn()
    ).getValues();
    data.forEach(row => {
      if (String(row[COL_STATUS - 1] || '').trim() !== 'Active') return;
      const qty = Number(row[ACC_COL_QTY - 1] || 0);
      if (qty <= ALERT_LOW_QTY) {
        lowItems.push({
          type: 'Accessory',
          name: String(row[3] || ''),
          size: '',
          qty,
        });
      }
    });
  }


  const devSheet = ss.getSheetByName(TABS.device);
  if (devSheet && devSheet.getLastRow() >= DATA_START) {
    const data = devSheet.getRange(
      DATA_START, 1,
      devSheet.getLastRow() - DATA_START + 1,
      devSheet.getLastColumn()
    ).getValues();
    data.forEach(row => {
      if (String(row[COL_STATUS - 1] || '').trim() !== 'Active') return;
      const expRaw = row[11]; // col 12 = Expiry
      if (!expRaw) return;
      const expDate = new Date(expRaw);
      if (isNaN(expDate.getTime())) return;
      expDate.setHours(0,0,0,0);
      const daysLeft = Math.ceil((expDate - today) / 86400000);
      if (daysLeft <= ALERT_EXP_DAYS) {
        expItems.push({
          type:    'Device',
          name:    String(row[3] || ''),
          sn:      String(row[DEV_COL_SN - 1] || ''),
          daysLeft,
          expDate: Utilities.formatDate(expDate, TZ, 'dd/MM/yyyy'),
        });
      }
    });
  }

  return { lowItems, expItems };
}

        Low-stock threshold: ${ALERT_LOW_QTY} items &nbsp;|&nbsp; Expiry warning: ${ALERT_EXP_DAYS} days<br>
        To change recipients, update ALERT_EMAIL in Apps Script.
      </div>
    </div>
  </div>
</body></html>`;
}

/* =========================================================
 * DUPLICATE CHECK
 * ========================================================= */
function checkDuplicate(ss, type, sn, lot, name, size) {
  return !!findDuplicate(ss, type, sn, lot, name, size);
}

function findDuplicate(ss, type, sn, lot, name, size) {
  const sheet = ss.getSheetByName(TABS[type]);
  if (!sheet) return null;
  const last = sheet.getLastRow();
  if (last < DATA_START) return null;

  const numCols = sheet.getLastColumn();
  const data = sheet.getRange(DATA_START, 1, last - DATA_START + 1, numCols).getValues();

  for (let i = 0; i < data.length; i++) {
    const row = data[i];
    const status = normalizeText(row[COL_STATUS - 1]);
    if (status !== 'Active') continue;

    if (type === 'device') {
      if (sn) {

        const rowSn = normalizeText(row[DEV_COL_SN - 1]);
        if (rowSn.toUpperCase() === normalizeText(sn).toUpperCase()) {
          return {
            row: i + DATA_START,
            status: status,
            sn: rowSn,
            lot: normalizeText(row[DEV_COL_LOT - 1]),
            name: normalizeText(row[3])
          };
        }
      }


    }



  }
  return null;
}
